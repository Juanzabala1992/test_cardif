die()

need_cmd()
{
    command -v "$1" >/dev/null 2>&1 || \

fingerprint_cert()
{
    openssl x509 \
        -in "$1" \
        -noout \
        -fingerprint \
        -sha256 |
    cut -d= -f2
}


# ============================================================
# YAML PREPARADO PARA RESTAURACION
# ============================================================

backup_restore_yaml()
{
    local KIND="$1"
    local NAME="$2"
    local OUTPUT="$3"

    oc get "$KIND" "$NAME" \
        -n "$NS" \
        -o yaml \
        --show-managed-fields=false |
    sed \
        -e '/^  creationTimestamp:/d' \
        -e '/^  resourceVersion:/d' \
        -e '/^  uid:/d' \
        -e '/^  generation:/d' |
    awk '
        /^status:/ {
            exit
        }

        {
            print
        }
    ' > "$OUTPUT"

    chmod 600 "$OUTPUT"
}


# ============================================================
# ROLLBACK
# ============================================================

restore_everything()
{
    echo
    echo "============================================================"
    echo " INICIANDO ROLLBACK"
    echo "============================================================"
    echo

    local RC=0

    shopt -s nullglob

    local SECRET_FILES=("$RESTORE_SECRETS"/*.yaml)
    local ROUTE_FILES=("$RESTORE_ROUTES"/*.yaml)

    echo "Restaurando Secrets..."
    echo

    for FILE in "${SECRET_FILES[@]}"
    do

        echo " -> $(basename "$FILE")"

        if ! oc apply \
            -n "$NS" \
            -f "$FILE"
        then
            RC=1
        fi

    done


    echo
    echo "Restaurando Routes..."
    echo

    for FILE in "${ROUTE_FILES[@]}"
    do

        echo " -> $(basename "$FILE")"

        if ! oc apply \
            -n "$NS" \
            -f "$FILE"
        then
            RC=1
        fi

    done


    echo

    if [ "$RC" -eq 0 ]; then

        echo "============================================================"
        echo " ROLLBACK FINALIZADO"
        echo "============================================================"

    else

        echo "============================================================"
        echo " ATENCION"
        echo "============================================================"
        echo
        echo "Alguno de los recursos no pudo restaurarse."
        echo
        echo "Los YAML permanecen en:"
        echo "$BACKUP_DIR"

    fi

    return "$RC"
}

test_routes()
{
    local OUTPUT="$1"

    {

        echo "Host: $HOST"
        echo

    } > "$OUTPUT"


    for ROUTE in "${ALL_ROUTES[@]}"
    do

        PATH_URL=$(
            oc get route "$ROUTE" \
                -n "$NS" \
                -o jsonpath='{.spec.path}' \
                2>/dev/null || true
        )


        [ -n "$PATH_URL" ] || PATH_URL="/"

        URL="https://${HOST}${PATH_URL}"


        RESULT=$(
            curl \
                -k \
                -sS \
                -o /dev/null \
                --connect-timeout 10 \
                --max-time 20 \
                -w "HTTP=%{http_code} TIME=%{time_total}" \
                "$URL" \
                2>/dev/null \
                || echo "HTTP=000 ERROR=CURL"
        )


        echo "${ROUTE}|${URL}|${RESULT}" >> "$OUTPUT"


        printf "%-32s %-55s %s\n" \
            "$ROUTE" \
            "$URL" \
            "$RESULT"

    done
}


# ============================================================
# CERTIFICADO REMOTO DEL HOST
# ============================================================

capture_remote_certificate()
{
    local OUTPUT="$1"

    local TMP_REMOTE="${TMP_DIR}/remote-cert.pem"

    rm -f "$TMP_REMOTE"


    if command -v timeout >/dev/null 2>&1; then

        if echo |
            timeout 15 \
            openssl s_client \
                -connect "${HOST}:443" \
                -servername "$HOST" \
                2>/dev/null |
            openssl x509 \
                > "$TMP_REMOTE" \
                2>/dev/null
        then

            openssl x509 \
                -in "$TMP_REMOTE" \
                -noout \
                -subject \
                -issuer \
                -serial \
                -dates \
                -fingerprint \
                -sha256 \
                > "$OUTPUT"

            return 0

        fi

    fi


    echo "No fue posible consultar certificado remoto." \
        > "$OUTPUT"

    return 0
}


# ============================================================
# REVISAR CERTIFICADOS MONTADOS EN PODS
# ============================================================

check_pod_certificates()
{
    local OUTPUT="$1"
    local COMPARE_NEW="$2"

    > "$OUTPUT"

    echo
    echo "============================================================"
    echo " CERTIFICADOS DENTRO DE LOS PODS"
    echo "============================================================"
    echo


    for SECRET in "${SECRETS_UPDATE[@]}"
    do

        echo "Secret: $SECRET" | tee -a "$OUTPUT"


        MOUNTS=$(
            oc get pods \
                -n "$NS" \
                -o json |
            python3 -c '
import json
import sys

secret_name=sys.argv[1]
data=json.load(sys.stdin)

for pod in data.get("items", []):

    pod_name=pod["metadata"]["name"]

    volumes={}

    for volume in pod.get("spec", {}).get("volumes", []):

        secret=volume.get("secret")

        if secret and secret.get("secretName") == secret_name:
            volumes[volume["name"]] = True


    if not volumes:
        continue


    for container in pod.get("spec", {}).get("containers", []):

        cname=container["name"]

        for mount in container.get("volumeMounts", []):

            if mount.get("name") not in volumes:
                continue

            mount_path=mount.get("mountPath", "")
            sub_path=mount.get("subPath", "")

            print(
                "|".join([
                    pod_name,
                    cname,
                    mount_path,
                    sub_path
                ])
            )
' "$SECRET"
        )


        if [ -z "$MOUNTS" ]; then

            echo "  No se encontraron pods montando este Secret." \
                | tee -a "$OUTPUT"

            echo | tee -a "$OUTPUT"

            continue

        fi


        while IFS='|' read -r POD CONTAINER MOUNT_PATH SUB_PATH
        do

            [ -n "$POD" ] || continue


            echo "  Pod       : $POD" \
                | tee -a "$OUTPUT"

            echo "  Container : $CONTAINER" \
                | tee -a "$OUTPUT"

            echo "  Mount     : $MOUNT_PATH" \
                | tee -a "$OUTPUT"


            CERT_PATH=""


            if [ -n "$SUB_PATH" ]; then

                echo "  subPath   : $SUB_PATH" \
                    | tee -a "$OUTPUT"


                if [ "$SUB_PATH" = "server.crt" ]; then

                    CERT_PATH="$MOUNT_PATH"

                    echo "  ATENCION: server.crt usa subPath." \
                        | tee -a "$OUTPUT"

                    echo "  Los Secret montados con subPath normalmente requieren" \
                        | tee -a "$OUTPUT"

                    echo "  recrear/reiniciar el pod para reflejar el cambio." \
                        | tee -a "$OUTPUT"

                else

                    echo "  Este mount no corresponde directamente a server.crt." \
                        | tee -a "$OUTPUT"

                    echo | tee -a "$OUTPUT"

                    continue

                fi

            else

                CERT_PATH="${MOUNT_PATH%/}/server.crt"

            fi


            POD_CERT="${TMP_DIR}/pod-${POD}-${CONTAINER}.crt"


            if oc exec \
                -n "$NS" \
                "$POD" \
                -c "$CONTAINER" \
                -- cat "$CERT_PATH" \
                > "$POD_CERT" \
                2>/dev/null
            then

                if openssl x509 \
                    -in "$POD_CERT" \
                    -noout \
                    >/dev/null 2>&1
                then

                    POD_FP=$(fingerprint_cert "$POD_CERT")


                    POD_DATES=$(
                        openssl x509 \
                            -in "$POD_CERT" \
                            -noout \
                            -dates |
                        tr '\n' ' '
                    )


                    echo "  Fingerprint: $POD_FP" \
                        | tee -a "$OUTPUT"

                    echo "  $POD_DATES" \
                        | tee -a "$OUTPUT"


                    if [ "$COMPARE_NEW" = "YES" ]; then

                        if [ "$POD_FP" = "$NEW_FP" ]; then

                            echo "  Resultado: CERTIFICADO NUEVO OK" \
                                | tee -a "$OUTPUT"

                        else

                            echo "  Resultado: TODAVIA TIENE CERTIFICADO ANTERIOR" \
                                | tee -a "$OUTPUT"

                        fi

                    fi

                else

                    echo "  No se pudo interpretar $CERT_PATH como certificado." \
                        | tee -a "$OUTPUT"

                fi

            else

                echo "  No fue posible leer $CERT_PATH desde el pod." \
                    | tee -a "$OUTPUT"

            fi


            echo | tee -a "$OUTPUT"

            rm -f "$POD_CERT"

        done <<< "$MOUNTS"

    done
}

# ============================================================

echo
echo "============================================================"
echo " BUSCANDO SECRETS *-file"
echo "============================================================"
echo


mapfile -t SECRET_CANDIDATES < <(
    oc get secrets \
        -n "$NS" \
        -o name |
    sed 's#^secret/##' |
    grep -E -- '-file$' || true
)


SECRETS_UPDATE=()

> "${INFO_DIR}/secrets-omitidos.txt"


for SECRET in "${SECRET_CANDIDATES[@]}"
do

    CRT_DATA=$(
        oc get secret "$SECRET" \
            -n "$NS" \
            -o jsonpath='{.data.server\.crt}' \
            2>/dev/null || true
    )

    KEY_DATA=$(
        oc get secret "$SECRET" \
            -n "$NS" \
            -o jsonpath='{.data.server\.key}' \
            2>/dev/null || true
    )


    if [ -z "$CRT_DATA" ]; then

        echo "$SECRET | no tiene server.crt" \
            >> "${INFO_DIR}/secrets-omitidos.txt"

        continue

    fi


    if [ -z "$KEY_DATA" ]; then

        echo "$SECRET | no tiene server.key" \
            >> "${INFO_DIR}/secrets-omitidos.txt"

        continue

    fi


    TEST_CERT="${TMP_DIR}/secret-${SECRET}.crt"


    if ! printf "%s" "$CRT_DATA" |
        base64 -d \
        > "$TEST_CERT" \
        2>/dev/null
    then

        echo "$SECRET | server.crt no se pudo decodificar" \
            >> "${INFO_DIR}/secrets-omitidos.txt"

        rm -f "$TEST_CERT"

        continue

    fi


    if openssl x509 \
        -in "$TEST_CERT" \
        -checkhost "$HOST" \
        -noout \
        >/dev/null 2>&1
    then

        SECRETS_UPDATE+=("$SECRET")

    else

        echo "$SECRET | certificado no corresponde a $HOST" \
            >> "${INFO_DIR}/secrets-omitidos.txt"

    fi


    rm -f "$TEST_CERT"

done


[ "${#SECRETS_UPDATE[@]}" -gt 0 ] \
    || die "No se encontraron Secrets *-file correspondientes a $HOST"


printf "%s\n" "${SECRETS_UPDATE[@]}" \
    > "${INFO_DIR}/secrets-actualizar.txt"


echo "SECRETS QUE SI SE ACTUALIZARAN:"
echo "------------------------------------------------------------"

printf "  %s\n" "${SECRETS_UPDATE[@]}"

echo "------------------------------------------------------------"


echo
echo "SECRETS *-file OMITIDOS:"
echo "------------------------------------------------------------"

if [ -s "${INFO_DIR}/secrets-omitidos.txt" ]; then

    cat "${INFO_DIR}/secrets-omitidos.txt"

else

    echo "  Ninguno"

fi

echo "------------------------------------------------------------"

# ============================================================
# GUARDAR HASH DE destinationCACertificate
# ============================================================

DEST_CA_HASH_FILE="${INFO_DIR}/destination-ca-original.sha256"

> "$DEST_CA_HASH_FILE"


for ROUTE in "${ROUTES_UPDATE[@]}"
do

    DEST_CA=$(
        oc get route "$ROUTE" \
            -n "$NS" \
            -o jsonpath='{.spec.tls.destinationCACertificate}' \
            2>/dev/null || true
    )


    if [ -n "$DEST_CA" ]; then

        DEST_HASH=$(
            printf "%s" "$DEST_CA" |
            sha256sum |
            awk '{print $1}'
        )

        echo "${ROUTE}|${DEST_HASH}" \
            >> "$DEST_CA_HASH_FILE"

    else

        echo "${ROUTE}|NONE" \
            >> "$DEST_CA_HASH_FILE"

    fi

done


# ============================================================
# VALIDAR CA ACTUAL
#
# SOLO EN ROUTES QUE REALMENTE TIENEN CERTIFICADO PROPIO
# ============================================================

if [ "$CA_MODE" = "CURRENT" ]; then

    echo
    echo "============================================================"
    echo " VALIDANDO caCertificate ACTUAL"
    echo "============================================================"
    echo


    CA_ERRORS=0


    for ROUTE in "${ROUTES_UPDATE[@]}"
    do

        CURRENT_CA=$(
            oc get route "$ROUTE" \
                -n "$NS" \
                -o jsonpath='{.spec.tls.caCertificate}' \
                2>/dev/null || true
        )


        if [ -z "$CURRENT_CA" ]; then

            echo "ERROR: $ROUTE no tiene caCertificate."

            CA_ERRORS=$((CA_ERRORS + 1))

            continue

        fi


        ROUTE_CA="${TMP_DIR}/ca-${ROUTE}.pem"


        printf "%s\n" "$CURRENT_CA" \
            > "$ROUTE_CA"


        if openssl verify \
            -partial_chain \
            -CAfile "$ROUTE_CA" \
            "$LEAF_CERT" \
            >/dev/null 2>&1
        then

            echo "OK: $ROUTE"

        else

            echo "ERROR: CA de $ROUTE no valida certificado nuevo."

            CA_ERRORS=$((CA_ERRORS + 1))

        fi

    done


    if [ "$CA_ERRORS" -gt 0 ]; then

        echo
        echo "============================================================"
        echo " NO SE REALIZARA NINGUN CAMBIO"
        echo "============================================================"
        echo
        echo "Hay Routes con certificado propio cuya CA"
        echo "no valida el nuevo certificado."
        echo
        echo "Backup:"
        echo "$BACKUP_DIR"

        exit 1

    fi

fi


# ============================================================
# GENERAR RESTORE.SH
# ============================================================

cat > "${BACKUP_DIR}/RESTORE.sh" <<EOF
#!/usr/bin/env bash

set -Eeuo pipefail

NS="$NS"

DIR="\$(cd "\$(dirname "\$0")" && pwd)"

ROUTES_DIR="\$DIR/restore/routes"
SECRETS_DIR="\$DIR/restore/secrets"

echo
echo "============================================================"
echo " RESTAURAR CERTIFICADOS"
echo "============================================================"
echo
echo "Namespace: \$NS"
echo "Backup:    \$DIR"
echo

read -r -p "Escribe RESTAURAR para continuar: " RESP

if [ "\$RESP" != "RESTAURAR" ]; then

    echo "Operacion cancelada."
    exit 0

fi


shopt -s nullglob

SECRET_FILES=("\$SECRETS_DIR"/*.yaml)
ROUTE_FILES=("\$ROUTES_DIR"/*.yaml)


echo
echo "Restaurando Secrets..."

for FILE in "\${SECRET_FILES[@]}"
do

    echo " -> \$(basename "\$FILE")"

    oc apply \
        -n "\$NS" \
        -f "\$FILE"

done


echo
echo "Restaurando Routes..."

for FILE in "\${ROUTE_FILES[@]}"
do

    echo " -> \$(basename "\$FILE")"

    oc apply \
        -n "\$NS" \
        -f "\$FILE"

done


echo
echo "============================================================"
echo " RESTAURACION FINALIZADA"
echo "============================================================"
EOF


chmod 700 "${BACKUP_DIR}/RESTORE.sh"


# ============================================================

# ============================================================

echo "============================================================"

else
    echo "  Ninguna"
fi


echo
echo "SECRETS QUE SE MODIFICARAN:"
printf "  %s\n" "${SECRETS_UPDATE[@]}"


echo
echo "CA MODE: $CA_MODE"

echo
echo "destinationCACertificate:"
echo "  NO SE MODIFICARA"

echo
echo "Backup:"
echo "  $BACKUP_DIR"

echo
echo "Rollback:"
echo "  $BACKUP_DIR/RESTORE.sh"

echo
echo "============================================================"
echo " HASTA ESTE MOMENTO NO SE HA MODIFICADO OPENSHIFT"
echo "============================================================"
echo


read -r -p "Escribe ACTUALIZAR para realizar el cambio: " RESP


if [ "$RESP" != "ACTUALIZAR" ]; then

    echo
    echo "Operacion cancelada."
    echo
    echo "No se modifico ningun Route ni Secret."
    echo
    echo "Backup:"
    echo "$BACKUP_DIR"

    exit 0

fi


# ============================================================
# PATCH DE ROUTES
# ============================================================

ROUTE_PATCH="${TMP_DIR}/route-patch.json"


python3 - \
    "$LEAF_CERT" \
    "$KEY" \
    "$NEW_CA_CHAIN" \
    "$CA_MODE" \
    > "$ROUTE_PATCH" <<'PY'

import json
import sys

cert_file=sys.argv[1]
key_file=sys.argv[2]
ca_file=sys.argv[3]
ca_mode=sys.argv[4]


with open(cert_file) as f:
    certificate=f.read()

with open(key_file) as f:
    key=f.read()


tls={
    "certificate": certificate,
    "key": key
}


if ca_mode in ("BUNDLE", "FILE"):

    with open(ca_file) as f:
        tls["caCertificate"]=f.read()


patch={
    "spec": {
        "tls": tls
    }
}


print(json.dumps(patch))
PY


chmod 600 "$ROUTE_PATCH"


# ============================================================
# PATCH DE SECRETS
# ============================================================

SECRET_PATCH="${TMP_DIR}/secret-patch.json"


python3 - \
    "$CERT" \
    "$KEY" \
    > "$SECRET_PATCH" <<'PY'

import json
import sys
import base64

cert_file=sys.argv[1]
key_file=sys.argv[2]


with open(cert_file, "rb") as f:
    cert=base64.b64encode(f.read()).decode("ascii")


with open(key_file, "rb") as f:
    key=base64.b64encode(f.read()).decode("ascii")


patch={
    "data": {
        "server.crt": cert,
        "server.key": key
    }
}


print(json.dumps(patch))
PY


chmod 600 "$SECRET_PATCH"

        -n "$NS" \
        --type=merge \
        --patch-file="$SECRET_PATCH"
    then

        echo
        echo "ERROR actualizando $SECRET"
        echo "Ejecutando rollback..."

        restore_everything || true

        exit 1

    fi

echo " VALIDANDO ROUTES"
echo "============================================================"
echo


ERRORS=0


for ROUTE in "${ROUTES_UPDATE[@]}"
do

    echo "Route: $ROUTE"


    CURRENT_CERT="${TMP_DIR}/route-${ROUTE}.crt"


    oc get route "$ROUTE" \
        -n "$NS" \
        -o jsonpath='{.spec.tls.certificate}' \
        > "$CURRENT_CERT"


    CURRENT_FP=$(fingerprint_cert "$CURRENT_CERT")


    if [ "$CURRENT_FP" = "$NEW_FP" ]; then

        echo "  certificate: OK"

    else

        echo "  certificate: ERROR"

        ERRORS=$((ERRORS + 1))

    fi


    OLD_DEST_HASH=$(
        awk -F'|' \
            -v route="$ROUTE" \
            '$1 == route {print $2}' \
            "$DEST_CA_HASH_FILE"
    )


    CURRENT_DEST_CA=$(
        oc get route "$ROUTE" \
            -n "$NS" \
            -o jsonpath='{.spec.tls.destinationCACertificate}' \
            2>/dev/null || true
    )


    if [ -n "$CURRENT_DEST_CA" ]; then

        CURRENT_DEST_HASH=$(
            printf "%s" "$CURRENT_DEST_CA" |
            sha256sum |
            awk '{print $1}'
        )

    else

        CURRENT_DEST_HASH="NONE"

    fi


    if [ "$OLD_DEST_HASH" = "$CURRENT_DEST_HASH" ]; then

        echo "  destinationCACertificate: SIN CAMBIOS"

    else

        echo "  destinationCACertificate: ERROR"

        ERRORS=$((ERRORS + 1))

    fi


    rm -f "$CURRENT_CERT"

done


# ============================================================
# VALIDAR SECRETS
# ============================================================

echo
echo "============================================================"
echo " VALIDANDO SECRETS"
echo "============================================================"
echo


for SECRET in "${SECRETS_UPDATE[@]}"
do

    echo "Secret: $SECRET"


    SECRET_CERT="${TMP_DIR}/${SECRET}.crt"


    if ! oc get secret "$SECRET" \
        -n "$NS" \
        -o jsonpath='{.data.server\.crt}' |
        base64 -d \
        > "$SECRET_CERT"
    then

        echo "  server.crt: ERROR"

        ERRORS=$((ERRORS + 1))

        continue

    fi


    SECRET_FP=$(fingerprint_cert "$SECRET_CERT")


    if [ "$SECRET_FP" = "$NEW_FP" ]; then

        echo "  server.crt: OK"

    else

        echo "  server.crt: ERROR"

        ERRORS=$((ERRORS + 1))

    fi


    SECRET_KEY_HASH=$(
        oc get secret "$SECRET" \
            -n "$NS" \
            -o jsonpath='{.data.server\.key}' |
        base64 -d |
        openssl pkey \
            -passin pass: \
            -pubout \
            -outform DER \
            2>/dev/null |
        sha256sum |
        awk '{print $1}'
    )


    if [ "$SECRET_KEY_HASH" = "$KEY_HASH" ]; then

        echo "  server.key: OK"

    else

        echo "  server.key: ERROR"

        ERRORS=$((ERRORS + 1))

    fi


    rm -f "$SECRET_CERT"

done
