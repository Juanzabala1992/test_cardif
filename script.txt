restore_everything()
{
    echo
    echo "============================================================"
    echo " INICIANDO ROLLBACK PRD"
    echo "============================================================"
    echo


    local RC=0

    shopt -s nullglob


    local SECRET_FILES=("$RESTORE_SECRETS"/*.yaml)
    local ROUTE_FILES=("$RESTORE_ROUTES"/*.yaml)


    echo "Restaurando Secrets..."
    echo


    for FILE in "${SECRET_FILES[@]}"
    do

        echo " -> $(basename "$FILE")"


        if ! oc apply \
            -n "$NS" \
            -f "$FILE"
        then

            RC=1

        fi

    done


    echo
    echo "Restaurando Routes..."
    echo


    for FILE in "${ROUTE_FILES[@]}"
    do

        echo " -> $(basename "$FILE")"


        if ! oc apply \
            -n "$NS" \
            -f "$FILE"
        then

            RC=1

        fi

    done


    echo


    if [ "$RC" -eq 0 ]; then

        echo "============================================================"
        echo " ROLLBACK PRD FINALIZADO"
        echo "============================================================"

    else

        echo "============================================================"
        echo " ATENCION"
        echo "============================================================"
        echo
        echo "Alguno de los recursos no pudo restaurarse."
        echo
        echo "Los YAML permanecen en:"
        echo "$BACKUP_DIR"

    fi


    return "$RC"
}

test_routes()
{
    local OUTPUT="$1"


    {
        echo "Fecha: $(date)"
        echo "Namespace: $NS"
        echo "Host: $HOST"
        echo

    } > "$OUTPUT"


    for ROUTE in "${ALL_ROUTES[@]}"
    do

        PATH_URL=$(
            oc get route "$ROUTE" \
                -n "$NS" \
                -o jsonpath='{.spec.path}' \
                2>/dev/null || true
        )


        [ -n "$PATH_URL" ] || PATH_URL="/"


        URL="https://${HOST}${PATH_URL}"


        RESULT=$(
            curl \
                -k \
                -sS \
                -o /dev/null \
                --connect-timeout 10 \
                --max-time 20 \
                -w "HTTP=%{http_code} TIME=%{time_total}" \
                "$URL" \
                2>/dev/null \
                || echo "HTTP=000 ERROR=CURL"
        )


        echo "${ROUTE}|${URL}|${RESULT}" >> "$OUTPUT"


        printf "%-32s %-60s %s\n" \
            "$ROUTE" \
            "$URL" \
            "$RESULT"

    done
}

capture_remote_certificate()
{
    local OUTPUT="$1"

    local TMP_REMOTE="${TMP_DIR}/remote-cert.pem"


    rm -f "$TMP_REMOTE"


    if command -v timeout >/dev/null 2>&1
    then

        if echo |
            timeout 15 \
            openssl s_client \
                -connect "${HOST}:443" \
                -servername "$HOST" \
                2>/dev/null |
            openssl x509 \
                > "$TMP_REMOTE" \
                2>/dev/null
        then

            openssl x509 \
                -in "$TMP_REMOTE" \
                -noout \
                -subject \
                -issuer \
                -serial \
                -dates \
                -fingerprint \
                -sha256 \
                > "$OUTPUT"

            return 0

        fi

    fi


    echo "No fue posible consultar certificado remoto." \
        > "$OUTPUT"

    return 0
}

# ============================================================

check_pod_certificates()
{
    local OUTPUT="$1"
    local COMPARE_NEW="$2"


    > "$OUTPUT"


    echo
    echo "============================================================"
    echo " CERTIFICADOS DENTRO DE PODS"
    echo "============================================================"
    echo


    for SECRET in "${SECRETS_UPDATE[@]}"
    do

        echo "Secret: $SECRET" | tee -a "$OUTPUT"


        MOUNTS=$(
            oc get pods \
                -n "$NS" \
                -o json |
            python3 -c '
import json
import sys

secret_name=sys.argv[1]
data=json.load(sys.stdin)


for pod in data.get("items", []):

    pod_name=pod.get("metadata", {}).get("name", "")

    volumes={}


    for volume in pod.get("spec", {}).get("volumes", []):

        secret=volume.get("secret")

        if not secret:
            continue

        if secret.get("secretName") != secret_name:
            continue


        cert_path="server.crt"


        for item in secret.get("items", []) or []:

            if item.get("key") == "server.crt":
                cert_path=item.get("path", "server.crt")


        volumes[volume["name"]]=cert_path


    if not volumes:
        continue


    for container in pod.get("spec", {}).get("containers", []):

        cname=container.get("name", "")


        for mount in container.get("volumeMounts", []):

            volume_name=mount.get("name")


            if volume_name not in volumes:
                continue


            mount_path=mount.get("mountPath", "")
            sub_path=mount.get("subPath", "")
            cert_relative=volumes[volume_name]


            print(
                "|".join([
                    pod_name,
                    cname,
                    mount_path,
                    sub_path,
                    cert_relative
                ])
            )
' "$SECRET"
        )


        if [ -z "$MOUNTS" ]; then

            echo "  No se encontraron pods montando este Secret." \
                | tee -a "$OUTPUT"

            echo | tee -a "$OUTPUT"

            continue

        fi


        while IFS='|' read -r POD CONTAINER MOUNT_PATH SUB_PATH CERT_REL
        do

            [ -n "$POD" ] || continue


            echo "  Pod       : $POD" | tee -a "$OUTPUT"
            echo "  Container : $CONTAINER" | tee -a "$OUTPUT"
            echo "  Mount     : $MOUNT_PATH" | tee -a "$OUTPUT"


            CERT_PATH=""


            if [ -n "$SUB_PATH" ]; then

                echo "  subPath   : $SUB_PATH" | tee -a "$OUTPUT"


                if [ "$SUB_PATH" = "$CERT_REL" ] || \
                   [ "$SUB_PATH" = "server.crt" ]
                then

                    CERT_PATH="$MOUNT_PATH"


                    echo "  ATENCION: certificado montado usando subPath." \
                        | tee -a "$OUTPUT"

                    echo "  Este pod puede requerir reinicio para reflejar" \
                        | tee -a "$OUTPUT"

                    echo "  modificaciones posteriores del Secret." \
                        | tee -a "$OUTPUT"

                else

                    echo "  El subPath no corresponde a server.crt." \
                        | tee -a "$OUTPUT"

                    echo | tee -a "$OUTPUT"

                    continue

                fi

            else

                CERT_PATH="${MOUNT_PATH%/}/${CERT_REL}"

            fi


            SAFE_POD=$(echo "$POD" | tr '/:' '__')
            SAFE_CONTAINER=$(echo "$CONTAINER" | tr '/:' '__')

            POD_CERT="${TMP_DIR}/pod-${SAFE_POD}-${SAFE_CONTAINER}.crt"


            if oc exec \
                -n "$NS" \
                "$POD" \
                -c "$CONTAINER" \
                -- cat "$CERT_PATH" \
                > "$POD_CERT" \
                2>/dev/null
            then

                if openssl x509 \
                    -in "$POD_CERT" \
                    -noout \
                    >/dev/null 2>&1
                then

                    POD_FP=$(fingerprint_cert "$POD_CERT")


                    POD_INFO=$(
                        openssl x509 \
                            -in "$POD_CERT" \
                            -noout \
                            -subject \
                            -issuer \
                            -dates |
                        tr '\n' ' '
                    )


                    echo "  $POD_INFO" | tee -a "$OUTPUT"

                    echo "  Fingerprint: $POD_FP" \
                        | tee -a "$OUTPUT"


                    if [ "$COMPARE_NEW" = "YES" ]; then

                        if [ "$POD_FP" = "$NEW_FP" ]; then

                            echo "  Resultado: CERTIFICADO NUEVO OK" \
                                | tee -a "$OUTPUT"

                        else

                            echo "  Resultado: TODAVIA TIENE CERTIFICADO ANTERIOR" \
                                | tee -a "$OUTPUT"

                        fi

                    fi

                else

                    echo "  No fue posible interpretar $CERT_PATH como certificado." \
                        | tee -a "$OUTPUT"

                fi

            else

                echo "  No fue posible leer $CERT_PATH desde el pod." \
                    | tee -a "$OUTPUT"

            fi


            echo | tee -a "$OUTPUT"

            rm -f "$POD_CERT"


        done <<< "$MOUNTS"

    done
}
