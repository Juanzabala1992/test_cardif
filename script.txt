CA_FILE=""

# ============================================================
# FUNCIONES

die()
{
    echo
    echo "ERROR: $*"
    echo
    exit 1
}


need_cmd()
{
    command -v "$1" >/dev/null 2>&1 || \
        die "No se encuentra el comando requerido: $1"
}


fingerprint_cert()
{
    openssl x509 \
        -in "$1" \
        -noout \
        -fingerprint \
        -sha256 |
    cut -d= -f2
}


backup_restore_yaml()
{
    local KIND="$1"
    local NAME="$2"
    local OUTPUT="$3"

    #
    # Genera una copia YAML apta para oc apply.
    #
    # Se eliminan campos administrados por Kubernetes que pueden
    # causar conflicto durante una restauracion.
    #

    oc get "$KIND" "$NAME" \
        -n "$NS" \
        -o yaml \
        --show-managed-fields=false |
    sed -E \
        -e '/^  creationTimestamp:/d' \
        -e '/^  resourceVersion:/d' \
        -e '/^  uid:/d' \
        -e '/^  generation:/d' |
    awk '
        /^status:/ {
            exit
        }

        {
            print
        }
    ' > "$OUTPUT"

    chmod 600 "$OUTPUT"
}


restore_everything()
{
    echo
    echo "============================================================"
    echo " INICIANDO ROLLBACK AUTOMATICO"
    echo "============================================================"
    echo

    local RC=0

    shopt -s nullglob

    local FILES_SECRET=("$RESTORE_SECRETS"/*.yaml)
    local FILES_ROUTE=("$RESTORE_ROUTES"/*.yaml)

    echo "Restaurando Secrets..."

    for FILE in "${FILES_SECRET[@]}"
    do
        echo "  -> $(basename "$FILE")"

        if ! oc apply -n "$NS" -f "$FILE"; then
            RC=1
        fi
    done


    echo
    echo "Restaurando Routes..."

    for FILE in "${FILES_ROUTE[@]}"
    do
        echo "  -> $(basename "$FILE")"

        if ! oc apply -n "$NS" -f "$FILE"; then
            RC=1
        fi
    done


    if [ "$RC" -eq 0 ]; then

        echo
        echo "ROLLBACK FINALIZADO."

    else

        echo
        echo "ATENCION:"
        echo "Alguno de los recursos no pudo restaurarse automaticamente."
        echo
        echo "Los YAML siguen disponibles en:"
        echo "$BACKUP_DIR"

    fi

    return "$RC"
}

    sha256sum |

        openssl x509 \
            -in "$PART" \
            -pubkey \
            -noout |
        openssl pkey \
            -pubin \
            -outform DER \
            2>/dev/null |
        sha256sum |
        awk '{print $1}'

    if [ "$PART_HASH" = "$KEY_HASH" ]; then

        if openssl x509 \
            -in "$PART" \
            -checkhost "$HOST" \
            -noout \
            >/dev/null 2>&1
        then

            LEAF_CERT="$PART"
            LEAF_MATCHES=$((LEAF_MATCHES + 1))

    fi

    if ! printf "%s" "$CRT_DATA" |

            >> "$SKIPPED_FILE"

    chmod 600 "$TEST_CERT"


    if ! openssl x509 \
        -in "$TEST_CERT" \
        -noout \
        >/dev/null 2>&1
    then

        echo "$SECRET | server.crt no es un certificado valido" \
            >> "$SKIPPED_FILE"

        rm -f "$TEST_CERT"

        continue

    fi


    if openssl x509 \
        -in "$TEST_CERT" \
        -checkhost "$HOST" \
        -noout \
        >/dev/null 2>&1
    then

        SECRETS+=("$SECRET")

    else

        echo "$SECRET | certificado no corresponde a $HOST" \
            >> "$SKIPPED_FILE"

    fi


    rm -f "$TEST_CERT"

done

with open(key_file, "r") as f:
    key = f.read()


tls = {
    "certificate": certificate,
    "key": key
}


#
# Solo cambiamos caCertificate si realmente tenemos
# una nueva cadena proveniente del CRT o un CA_FILE.
#
# Si usamos CURRENT, caCertificate queda sin tocar.
#

if ca_mode in ("BUNDLE", "FILE"):

    with open(ca_file, "r") as f:
        tls["caCertificate"] = f.read()


patch = {
    "spec": {
        "tls": tls
    }
}


print(
    json.dumps(patch)
)
PY


chmod 600 "$ROUTE_PATCH"


# ============================================================
# CREAR PATCH PARA SECRETS
#
# IMPORTANTE:
#
# En los Secrets SI guardamos server.crt COMPLETO tal como
# fue entregado. Si contiene la cadena, queda completa.
#
# ============================================================

SECRET_PATCH="${TMP_DIR}/secret-patch.json"


python3 - \
    "$CERT" \
    "$KEY" \
    > "$SECRET_PATCH" <<'PY'

import json
import sys
import base64


cert_file = sys.argv[1]
key_file = sys.argv[2]


with open(cert_file, "rb") as f:
    cert = base64.b64encode(
        f.read()
    ).decode("ascii")


with open(key_file, "rb") as f:
    key = base64.b64encode(
        f.read()
    ).decode("ascii")


patch = {
    "data": {
        "server.crt": cert,
        "server.key": key
    }
}


print(
    json.dumps(patch)
)
PY


chmod 600 "$SECRET_PATCH"


# ============================================================
# ACTUALIZAR ROUTES
# ============================================================

echo
echo "============================================================"
echo " ACTUALIZANDO ROUTES"
echo "============================================================"
echo


for ROUTE in "${ROUTES[@]}"
do

    echo "Actualizando Route: $ROUTE"

    if ! oc patch route "$ROUTE" \
        -n "$NS" \
        --type=merge \
        --patch-file="$ROUTE_PATCH"
    then

        echo
        echo "Fallo actualizando Route $ROUTE"
        echo "Se iniciara rollback."

        restore_everything || true

        exit 1

    fi

done


# ============================================================
# ACTUALIZAR SECRETS *-file
# ============================================================

echo
echo "============================================================"
echo " ACTUALIZANDO SECRETS *-file"
echo "============================================================"
echo


for SECRET in "${SECRETS[@]}"
do

    echo "Actualizando Secret: $SECRET"

    if ! oc patch secret "$SECRET" \
        -n "$NS" \
        --type=merge \
        --patch-file="$SECRET_PATCH"
    then

        echo
        echo "Fallo actualizando Secret $SECRET"
        echo "Se iniciara rollback."

        restore_everything || true

        exit 1

    fi

done


# ============================================================
# ELIMINAR PATCHES QUE CONTIENEN LA LLAVE NUEVA
# ============================================================

rm -f "$ROUTE_PATCH"
rm -f "$SECRET_PATCH"


# ============================================================
# VALIDAR CERTIFICADO NUEVO
# ============================================================

NEW_FP=$(fingerprint_cert "$LEAF_CERT")

ERRORS=0


echo
echo "============================================================"
echo " VALIDANDO ROUTES"
echo "============================================================"
echo


for ROUTE in "${ROUTES[@]}"
do

    echo "Route: $ROUTE"


    CURRENT_CERT="${TMP_DIR}/validate-route-${ROUTE}.crt"


    oc get route "$ROUTE" \
        -n "$NS" \
        -o jsonpath='{.spec.tls.certificate}' \
        > "$CURRENT_CERT"


    CURRENT_FP=$(fingerprint_cert "$CURRENT_CERT")


    if [ "$CURRENT_FP" = "$NEW_FP" ]; then

        echo "  certificate: OK"

    else

        echo "  certificate: ERROR"

        ERRORS=$((ERRORS + 1))

    fi


    # --------------------------------------------------------
    # VALIDAR QUE destinationCACertificate NO CAMBIO
    # --------------------------------------------------------

    OLD_DEST_HASH=$(
        awk -F'|' \
            -v route="$ROUTE" \
            '$1 == route {print $2}' \
            "$DEST_CA_HASH_FILE"
    )


    CURRENT_DEST_CA=$(
        oc get route "$ROUTE" \
            -n "$NS" \
            -o jsonpath='{.spec.tls.destinationCACertificate}' \
            2>/dev/null || true
    )


    if [ -n "$CURRENT_DEST_CA" ]; then

        CURRENT_DEST_HASH=$(
            printf "%s" "$CURRENT_DEST_CA" |
            sha256sum |
            awk '{print $1}'
        )

    else

        CURRENT_DEST_HASH="NONE"

    fi


    if [ "$OLD_DEST_HASH" = "$CURRENT_DEST_HASH" ]; then

        echo "  destinationCACertificate: SIN CAMBIOS"

    else

        echo "  destinationCACertificate: ERROR - CAMBIO DETECTADO"

        ERRORS=$((ERRORS + 1))

    fi


    rm -f "$CURRENT_CERT"

done


# ============================================================
# VALIDAR SECRETS
# ============================================================

echo
echo "============================================================"
echo " VALIDANDO SECRETS"
echo "============================================================"
echo


for SECRET in "${SECRETS[@]}"
do

    echo "Secret: $SECRET"


    CURRENT_SECRET_CERT="${TMP_DIR}/validate-${SECRET}.crt"


    if ! oc get secret "$SECRET" \
        -n "$NS" \
        -o jsonpath='{.data.server\.crt}' |
        base64 -d \
        > "$CURRENT_SECRET_CERT"
    then

        echo "  ERROR leyendo server.crt"

        ERRORS=$((ERRORS + 1))

        continue

    fi


    CURRENT_FP=$(fingerprint_cert "$CURRENT_SECRET_CERT")


    if [ "$CURRENT_FP" = "$NEW_FP" ]; then

        echo "  server.crt: OK"

    else

        echo "  server.crt: ERROR"

        ERRORS=$((ERRORS + 1))

    fi


    #
    # Comprobar tambien que server.key corresponde al CRT
    #

    SECRET_KEY_HASH=$(
        oc get secret "$SECRET" \
            -n "$NS" \
            -o jsonpath='{.data.server\.key}' |
        base64 -d |
        openssl pkey \
            -passin pass: \
            -pubout \
            -outform DER \
            2>/dev/null |
        sha256sum |
        awk '{print $1}'
    )


    if [ "$SECRET_KEY_HASH" = "$KEY_HASH" ]; then

        echo "  server.key: OK"

    else

        echo "  server.key: ERROR"

        ERRORS=$((ERRORS + 1))

    fi


    rm -f "$CURRENT_SECRET_CERT"

done


# ============================================================
# SI FALLA VALIDACION -> ROLLBACK AUTOMATICO
# ============================================================

if [ "$ERRORS" -gt 0 ]; then

    echo
    echo "============================================================"
    echo " SE DETECTARON $ERRORS ERRORES"
    echo "============================================================"
    echo
    echo "Se restaurara automaticamente el backup."

    restore_everything || true

    exit 1

fi


# ============================================================
# PRUEBA EXTERNA OPCIONAL DEL HOST
# ============================================================

echo
echo "============================================================"
echo " VALIDACION TLS DEL HOST"
echo "============================================================"
echo


if command -v timeout >/dev/null 2>&1; then

    REMOTE_CERT="${TMP_DIR}/remote-cert.pem"


    if echo |
        timeout 10 \
        openssl s_client \
            -connect "${HOST}:443" \
            -servername "$HOST" \
            2>/dev/null |
        openssl x509 \
            > "$REMOTE_CERT" \
            2>/dev/null
    then

        REMOTE_FP=$(fingerprint_cert "$REMOTE_CERT")


        if [ "$REMOTE_FP" = "$NEW_FP" ]; then

            echo "$HOST esta presentando el certificado NUEVO."
            echo "TLS externo: OK"

        else

            echo "ATENCION:"
            echo "$HOST responde, pero todavia presenta otro certificado."
            echo
            echo "Esto puede deberse a propagacion del router,"
            echo "balanceadores/proxy externos o cache intermedio."

        fi

    else

        echo "No fue posible validar $HOST:443 desde este bastion."
        echo "Esto NO genera rollback porque puede ser una restriccion de red."

    fi


    rm -f "$REMOTE_CERT"

else

    echo "El comando timeout no esta disponible."
    echo "Se omite la prueba TLS externa."

fi

# LIMPIAR TEMPORALES
# ============================================================

rm -rf "$TMP_DIR"

echo "============================================================"

echo "  $HOST"
echo

echo "  ${#ROUTES[@]}"
echo
echo "Secrets *-file actualizados:"
echo "  ${#SECRETS[@]}"
echo

echo "  $BACKUP_DIR"
echo
