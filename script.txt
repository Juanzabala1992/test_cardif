die()
{
    echo
    echo "============================================================"
    echo " ERROR"
    echo "============================================================"
    echo "$*"
    echo
    exit 1
}


need_cmd()
{
    command -v "$1" >/dev/null 2>&1 || \
        die "No se encuentra el comando requerido: $1"
}


fingerprint_cert()
{
    openssl x509 \
        -in "$1" \
        -noout \
        -fingerprint \
        -sha256 |
    cut -d= -f2
}


# ============================================================
# GENERAR YAML LIMPIO PARA RESTAURACION
# ============================================================

backup_restore_yaml()
{
    local KIND="$1"
    local NAME="$2"
    local OUTPUT="$3"

    oc get "$KIND" "$NAME" \
        -n "$NS" \
        -o yaml \
        --show-managed-fields=false |
    sed \
        -e '/^  creationTimestamp:/d' \
        -e '/^  resourceVersion:/d' \
        -e '/^  uid:/d' \
        -e '/^  generation:/d' |
    awk '
        /^status:/ {
            exit
        }

        {
            print
        }
    ' > "$OUTPUT"

    chmod 600 "$OUTPUT"
}


# ============================================================
# ROLLBACK AUTOMATICO
# ============================================================

restore_everything()
{
    echo
    echo "============================================================"
    echo " INICIANDO ROLLBACK"
    echo "============================================================"
    echo

    local RC=0

    shopt -s nullglob

    local FILES_SECRET=("$RESTORE_SECRETS"/*.yaml)
    local FILES_ROUTE=("$RESTORE_ROUTES"/*.yaml)

    echo "Restaurando Secrets..."
    echo

    for FILE in "${FILES_SECRET[@]}"
    do

        echo "Restaurando: $(basename "$FILE")"

        if ! oc apply \
            -n "$NS" \
            -f "$FILE"
        then

            RC=1

        fi

    done


    echo
    echo "Restaurando Routes..."
    echo

    for FILE in "${FILES_ROUTE[@]}"
    do

        echo "Restaurando: $(basename "$FILE")"

        if ! oc apply \
            -n "$NS" \
            -f "$FILE"
        then

            RC=1

        fi

    done


    if [ "$RC" -eq 0 ]; then

        echo
        echo "============================================================"
        echo " ROLLBACK FINALIZADO"
        echo "============================================================"

    else

        echo
        echo "============================================================"
        echo " ATENCION"
        echo "============================================================"
        echo
        echo "Alguno de los recursos no pudo restaurarse automaticamente."
        echo
        echo "Los YAML siguen disponibles en:"
        echo
        echo "$BACKUP_DIR"

    fi

    return "$RC"
}


# ============================================================
# PROBAR TODAS LAS URL DE LAS ROUTES
# ============================================================

test_routes()
{
    local OUTPUT="$1"

    {
        echo "============================================================"
        echo " PRUEBA DE ROUTES"
        echo "============================================================"
        echo
        echo "Fecha: $(date)"
        echo
        echo "Host: $HOST"
        echo

    } > "$OUTPUT"


    for ROUTE in "${ROUTES[@]}"
    do

        PATH_URL=$(
            oc get route "$ROUTE" \
                -n "$NS" \
                -o jsonpath='{.spec.path}' \
                2>/dev/null || true
        )


        if [ -z "$PATH_URL" ]; then
            PATH_URL="/"
        fi


        URL="https://${HOST}${PATH_URL}"


        RESULT=$(
            curl \
                -k \
                -sS \
                -o /dev/null \
                --connect-timeout 10 \
                --max-time 20 \
                -w "HTTP=%{http_code} TIME=%{time_total}" \
                "$URL" \
                2>/dev/null || echo "HTTP=000 ERROR=CURL"
        )


        {
            echo "ROUTE=$ROUTE"
            echo "URL=$URL"
            echo "$RESULT"
            echo

        } >> "$OUTPUT"


        printf "%-35s %-55s %s\n" \
            "$ROUTE" \
            "$URL" \
            "$RESULT"

    done
}


# ============================================================
# CAPTURAR CERTIFICADO QUE PRESENTA LA URL
# ============================================================

capture_remote_certificate()
{
    local OUTPUT="$1"

    REMOTE_TMP="${TMP_DIR}/remote.pem"

    rm -f "$REMOTE_TMP"

    if command -v timeout >/dev/null 2>&1; then

        if echo |
            timeout 15 \
            openssl s_client \
                -connect "${HOST}:443" \
                -servername "$HOST" \
                2>/dev/null |
            openssl x509 \
                > "$REMOTE_TMP" \
                2>/dev/null
        then

            openssl x509 \
                -in "$REMOTE_TMP" \
                -noout \
                -subject \
                -issuer \
                -serial \
                -dates \
                -fingerprint \
                -sha256 \
                > "$OUTPUT"

            return 0

        fi

    fi


    echo "No fue posible consultar el certificado remoto." \
        > "$OUTPUT"

    return 0
}

    for ROUTE in "${ROUTES[@]}"
    do

        CURRENT_CA=$(
            oc get route "$ROUTE" \
                -n "$NS" \

        )


        if [ -z "$CURRENT_CA" ]; then

            echo "ERROR: $ROUTE no tiene caCertificate"

            CA_ERRORS=$((CA_ERRORS + 1))

        printf "%s\n" "$CURRENT_CA" \

            echo "ERROR: La CA de $ROUTE no valida el certificado nuevo."

            CA_ERRORS=$((CA_ERRORS + 1))

        fi

    done


    if [ "$CA_ERRORS" -gt 0 ]; then

        echo
        echo "============================================================"
        echo " NO SE REALIZARA NINGUN CAMBIO"
        echo "============================================================"
        echo
        echo "La CA actual no valida correctamente el nuevo certificado."

        echo "Backup disponible en:"

    fi

import json

    )


    if [ -n "$CURRENT_DEST_CA" ]; then

            printf "%s" "$CURRENT_DEST_CA" |

            awk '{print $1}'
        )

    else

        CURRENT_DEST_HASH="NONE"

    fi


    if [ "$OLD_DEST_HASH" = "$CURRENT_DEST_HASH" ]; then

        echo "  destinationCACertificate: SIN CAMBIOS"

    else

        echo "  destinationCACertificate: ERROR"

        ERRORS=$((ERRORS + 1))

done
