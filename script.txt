#!/usr/bin/env bash
# Auditoria de consumo y errores en OpenShift DEV - SOLO LECTURA.
# Requisitos: bash, oc (logueado), awk, sort, sed, grep, mktemp y date.
# No requiere Python, jq, sudo ni permisos de escritura en el cluster.
# Uso: bash auditoria_workers_dev.sh
#      CPU_MIN_M=10 MEM_MIN_MI=10 REINICIOS_MAX=100 bash auditoria_workers_dev.sh
#      MODO=alguno bash auditoria_workers_dev.sh  # CPU > umbral O memoria > umbral
set -euo pipefail
export LC_ALL=C

CPU_MIN_M="${CPU_MIN_M:-10}"
MEM_MIN_MI="${MEM_MIN_MI:-10}"
REINICIOS_MAX="${REINICIOS_MAX:-100}"
MODO="${MODO:-ambos}"  # ambos = CPU > 10m Y memoria > 10Mi

for cmd in oc awk sort sed grep mktemp date; do
  if ! command -v "$cmd" >/dev/null 2>&1; then
    echo "ERROR: falta el comando requerido: $cmd" >&2
    exit 1
  fi
done
if ! [[ "$CPU_MIN_M" =~ ^[0-9]+([.][0-9]+)?$ && "$MEM_MIN_MI" =~ ^[0-9]+([.][0-9]+)?$ && "$REINICIOS_MAX" =~ ^[0-9]+$ ]]; then
  echo "ERROR: umbrales invalidos. Ejemplo: CPU_MIN_M=10 MEM_MIN_MI=10 REINICIOS_MAX=100" >&2
  exit 1
fi
if [[ "$MODO" != "ambos" && "$MODO" != "alguno" ]]; then
  echo "ERROR: MODO debe ser 'ambos' o 'alguno'" >&2
  exit 1
fi

SALIDA="${SALIDA:-$(pwd)/auditoria_dev_$(date +%Y%m%d_%H%M%S)}"
mkdir -p "$SALIDA"
TEMP="$(mktemp -d)"
trap 'rm -rf "$TEMP"' EXIT

echo "=== AUDITORIA OPENSHIFT DEV (SOLO LECTURA) ==="
echo "Contexto: $(oc config current-context 2>/dev/null || echo 'No disponible')"
echo "Resultados: $SALIDA"

# Los nodos workerdev-mashery* se incluyen: el filtro redis/mashery se aplica
# a SERVICIOS en el informe de consumo, no a nodos.
echo '[1/5] Buscando nodos workerdev...'
if ! oc get nodes -o jsonpath='{range .items[*]}{.metadata.name}{"\n"}{end}' > "$TEMP/nodos_todos.txt"; then
  echo 'ERROR: no se pudo consultar oc get nodes.' >&2
  exit 1
fi
awk 'tolower($0) ~ /^workerdev/' "$TEMP/nodos_todos.txt" | sort -u > "$TEMP/nodos_dev.txt"
if [[ ! -s "$TEMP/nodos_dev.txt" ]]; then
  echo 'ERROR: no se encontraron nodos workerdev (incluidos workerdev-mashery).' >&2
  exit 1
fi

echo '[2/5] Consultando pods de todos los namespaces...'
# Una fila por pod; los campos de estados de contenedores se separan con ;
# y los subcampos internos con : (son nombres/razones de Kubernetes).
POD_JSONPATH='{range .items[*]}{.metadata.namespace}{"\t"}{.metadata.name}{"\t"}{.spec.nodeName}{"\t"}{.status.phase}{"\t"}{.status.reason}{"\t"}{.metadata.ownerReferences[0].kind}{"\t"}{.metadata.ownerReferences[0].name}{"\t"}{range .status.containerStatuses[*]}{.name}{":"}{.restartCount}{":"}{.state.waiting.reason}{":"}{.state.terminated.reason}{":"}{.lastState.terminated.reason}{";"}{end}{"\t"}{range .status.initContainerStatuses[*]}{.name}{":"}{.restartCount}{":"}{.state.waiting.reason}{":"}{.state.terminated.reason}{":"}{.lastState.terminated.reason}{";"}{end}{"\t"}{range .status.conditions[*]}{.type}{":"}{.status}{":"}{.reason}{";"}{end}{"\n"}{end}'
if ! oc get pods -A -o "jsonpath=$POD_JSONPATH" > "$TEMP/pods_todos.tsv"; then
  echo 'ERROR: no se pudieron consultar pods de todos los namespaces.' >&2
  exit 1
fi
# Contiene c-dev en cualquier posicion, p. ej. lam-c-dev y per-c-dev-app.
awk -F '\t' 'tolower($1) ~ /c-dev/' "$TEMP/pods_todos.tsv" > "$TEMP/pods_dev.tsv"

echo '[3/5] Asociando ReplicaSets con su Deployment...'
RS_JSONPATH='{range .items[*]}{.metadata.namespace}{"\t"}{.metadata.name}{"\t"}{.metadata.ownerReferences[0].kind}{"\t"}{.metadata.ownerReferences[0].name}{"\n"}{end}'
if ! oc get replicasets -A -o "jsonpath=$RS_JSONPATH" > "$TEMP/rs.tsv"; then
  echo 'AVISO: sin permiso para listar ReplicaSets; se mostrara ReplicaSet en vez del Deployment.' >&2
  : > "$TEMP/rs.tsv"
fi

echo '[4/5] Leyendo consumo REAL de CPU y memoria (oc adm top pods -A)...'
METRICAS_OK=si
if ! oc adm top pods -A > "$TEMP/top.txt" 2> "$TEMP/metricas_error.txt"; then
  METRICAS_OK=no
  : > "$TEMP/top.txt"
  echo 'AVISO: no se pudieron obtener metricas; informes de consumo vacios y pods sin metricas listados.' >&2
  sed -n '1,3p' "$TEMP/metricas_error.txt" >&2
fi

# 01: consumo individual, 02: agrupado por deployment/workload.
# La agrupacion SUMA todas las replicas con metricas en workerdev, no solo
# las replicas que superaron el umbral individual.
awk -F '\t' -v OFS=';' -v cpuLim="$CPU_MIN_M" -v memLim="$MEM_MIN_MI" -v modo="$MODO" \
    -v podsOut="$TEMP/consumo_pods.tmp" -v depOut="$TEMP/consumo_deployments.tmp" \
    -v missOut="$TEMP/sin_metricas.tmp" \
    -f /dev/stdin "$TEMP/nodos_dev.txt" "$TEMP/rs.tsv" "$TEMP/pods_dev.tsv" "$TEMP/top.txt" <<'AWK'
function cpu_m(s, v) {
  if (s ~ /m$/) {sub(/m$/, "", s); return s+0}
  if (s ~ /u$/) {sub(/u$/, "", s); return (s+0)/1000}
  if (s ~ /n$/) {sub(/n$/, "", s); return (s+0)/1000000}
  if (s ~ /^[0-9]+([.][0-9]+)?$/) return (s+0)*1000
  return -1
}
function mem_mi(s) {
  if (s ~ /Ki$/) {sub(/Ki$/, "", s); return (s+0)/1024}
  if (s ~ /Mi$/) {sub(/Mi$/, "", s); return s+0}
