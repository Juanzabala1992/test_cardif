#!/usr/bin/env bash
# OpenShift: limits por microservicio y cuotas limits.* por namespace.
# Solo lectura. Requisitos: bash, oc, awk, sort (sin jq ni Python).
set -euo pipefail

SERVICIOS=(
  'eligibility-api'
  'eligibility-cancellation-policy-legacy-event-capture'
  'eligibility-event-capture'
  'eligibility-issued-policy-legacy-event-capture'
  'notification-center'
  'claims-document-service'
  'customer-portal'
  'customer-portal-backend'
)

SALIDA="${1:-limites_microservicios.csv}"
for cmd in oc awk sort; do
  command -v "$cmd" >/dev/null 2>&1 || { echo "ERROR: Falta $cmd" >&2; exit 1; }
done
oc whoami >/dev/null 2>&1 || { echo 'ERROR: Primero ejecuta oc login.' >&2; exit 1; }

tmpdir="$(mktemp -d)"
trap 'rm -rf "$tmpdir"' EXIT
workloads="$tmpdir/workloads.tsv"
quotas="$tmpdir/quotas.tsv"
filas="$tmpdir/filas.tsv"
enriquecido="$tmpdir/enriquecido.tsv"
: > "$workloads"; : > "$quotas"; : > "$filas"

# Datos por contenedor: namespace, servicio, tipo, replicas, contenedor, CPU limit, memoria limit.
plantilla_workload='{{range .items}}{{$w := .}}{{range .spec.template.spec.containers}}{{printf "%s\t%s\t%s\t%v\t%s\t" $w.metadata.namespace $w.metadata.name $w.kind $w.spec.replicas .name}}{{with .resources.limits}}{{with index . "cpu"}}{{.}}{{else}}SIN_LIMIT{{end}}{{else}}SIN_LIMIT{{end}}{{printf "\t"}}{{with .resources.limits}}{{with index . "memory"}}{{.}}{{else}}SIN_LIMIT{{end}}{{else}}SIN_LIMIT{{end}}{{printf "\n"}}{{end}}{{end}}'

oc get deployments -A -o "go-template=$plantilla_workload" > "$workloads" || {
  echo 'ERROR: No fue posible consultar los Deployments.' >&2; exit 1;
}
if ! oc get deploymentconfigs -A -o "go-template=$plantilla_workload" >> "$workloads" 2>/dev/null; then
  echo 'AVISO: No fue posible consultar DeploymentConfigs; se incluyen Deployments.' >&2
fi

# ResourceQuota, .spec.hard: limite total configurado en el namespace.
# El nombre de la cuota NO se imprimira como parte de los valores.
plantilla_quota='{{range .items}}{{printf "%s\t%s\t" .metadata.namespace .metadata.name}}{{with .spec.hard}}{{with index . "limits.cpu"}}{{.}}{{else}}SIN_CUOTA{{end}}{{else}}SIN_CUOTA{{end}}{{printf "\t"}}{{with .spec.hard}}{{with index . "limits.memory"}}{{.}}{{else}}SIN_CUOTA{{end}}{{else}}SIN_CUOTA{{end}}{{printf "\n"}}{{end}}'
quota_ok=1
if ! oc get resourcequotas -A -o "go-template=$plantilla_quota" > "$quotas"; then
  echo 'AVISO: No se pudieron consultar ResourceQuotas (permisos u otro error).' >&2
  quota_ok=0
  : > "$quotas"
fi

# for obligatorio: cada uno de los microservicios de la lista.
for servicio in "${SERVICIOS[@]}"; do
  awk -F '\t' -v OFS='\t' -v objetivo="$servicio" '
    function cpu_m(x) {
      if (x ~ /n$/) return (x+0)/1000000
      if (x ~ /u$/) return (x+0)/1000
      if (x ~ /m$/) return x+0
      return (x+0)*1000
    }
    function mem_mi(x,n) {
      n=x+0
      if (x ~ /Ki$/) return n/1024
      if (x ~ /Mi$/) return n
      if (x ~ /Gi$/) return n*1024
      if (x ~ /Ti$/) return n*1048576
      if (x ~ /Pi$/) return n*1073741824
      if (x ~ /Ei$/) return n*1099511627776
      if (x ~ /k$/) return n*1000/1048576
      if (x ~ /M$/) return n*1000000/1048576
      if (x ~ /G$/) return n*1000000000/1048576
      if (x ~ /T$/) return n*1000000000000/1048576
      return n/1048576
    }
    $2==objetivo {
      k=$1 SUBSEP $2 SUBSEP $3
      if (!(k in existe)) {
        existe[k]=1; ns[k]=$1; svc[k]=$2; tipo[k]=$3
        replicas[k]=($4=="<nil>" || $4=="<no value>" ? "1" : $4)
      }
      ncont[k]++
      if ($6=="SIN_LIMIT" || $6=="<no value>" || $6=="") sincpu[k]=1
      else { totalcpu[k]+=cpu_m($6); originalcpu[k]=$6 }
      if ($7=="SIN_LIMIT" || $7=="<no value>" || $7=="") sinmem[k]=1
      else { totalmem[k]+=mem_mi($7); originalmem[k]=$7 }
    }
    END {
      for (k in existe) {
        cpu=sincpu[k] ? "SIN_LIMIT" : (ncont[k]==1 ? originalcpu[k] : sprintf("%gm",totalcpu[k]))
        mem=sinmem[k] ? "SIN_LIMIT" : (ncont[k]==1 ? originalmem[k] : sprintf("%gMi",totalmem[k]))
        print ns[k],svc[k],tipo[k],replicas[k],cpu,mem
      }
    }
  ' "$workloads" | sort -t $'\t' -k1,1 > "$tmpdir/un_servicio.tsv"

  if [[ -s "$tmpdir/un_servicio.tsv" ]]; then
    cat "$tmpdir/un_servicio.tsv" >> "$filas"
  else
    printf 'NO_ENCONTRADO\t%s\t-\t-\t-\t-\n' "$servicio" >> "$filas"
    printf 'AVISO: No se encontro Deployment/DeploymentConfig llamado %s\n' "$servicio" >&2
  fi
done

# Anade los limites de namespace SIN el nombre del objeto ResourceQuota.
# Si hay varias ResourceQuota con limits.cpu/memory, muestra TODOS los valores,
# separados con ' / ' para no sumar cuotas que pueden tener scopes distintos.
awk -F '\t' -v OFS='\t' -v QUOTA_OK="$quota_ok" '
  FILENAME==ARGV[1] {
    if ($3!="SIN_CUOTA" && $3!="") {
      if ($1 in qcpu) qcpu[$1]=qcpu[$1] " / " $3; else qcpu[$1]=$3
    }
    if ($4!="SIN_CUOTA" && $4!="") {
      if ($1 in qmem) qmem[$1]=qmem[$1] " / " $4; else qmem[$1]=$4
    }
    next
  }
  {
    ns=$1
    if (ns=="NO_ENCONTRADO") {cpu="-";mem="-"}
    else if (QUOTA_OK==0) {cpu="NO_CONSULTADO";mem="NO_CONSULTADO"}
    else {
      cpu=(ns in qcpu ? qcpu[ns] : "SIN_CUOTA")
      mem=(ns in qmem ? qmem[ns] : "SIN_CUOTA")
    }
    print $1,$2,$3,$4,$5,$6,cpu,mem
  }
' "$quotas" "$filas" > "$enriquecido"

# CSV compatible con Excel en configuracion regional de Colombia: separador ';'.
awk -F '\t' -v OFS=';' '
  function quoted(x) {gsub(/"/,"\"\"",x);return "\"" x "\""}
  BEGIN {print "NAMESPACE;MICROSERVICIO;TIPO;REPLICAS;CPU_LIMIT_SERVICIO;MEM_LIMIT_SERVICIO;CPU_LIMIT_NAMESPACE;MEM_LIMIT_NAMESPACE"}
  {for(i=1;i<=8;i++) $i=quoted($i); print $1,$2,$3,$4,$5,$6,$7,$8}
' "$enriquecido" > "$SALIDA"

# CONSOLA: una tabla corta por servicio. El nombre largo del servicio es titulo,
# no una columna que provoque salto de linea.
awk -F '\t' '
  FNR==NR {
    w=length($1)
    if (w>max_ns[$2]) max_ns[$2]=w
    next
  }
  function barra(n,i) {for(i=0;i<n;i++) printf "-";printf "\n"}
  $2!=anterior {
    ancho=max_ns[$2]; if(ancho<12) ancho=12
    printf "\nMICROSERVICIO: %s\n",$2
    printf "%-*s %4s %9s %10s %9s %10s\n",ancho,"NAMESPACE","REP","CPU_SVC","MEM_SVC","CPU_NS","MEM_NS"
    barra(ancho+49)
    anterior=$2
  }
  {
    ancho=max_ns[$2]; if(ancho<12) ancho=12
    printf "%-*s %4s %9s %10s %9s %10s\n",ancho,$1,$4,$5,$6,$7,$8
  }
' "$enriquecido" "$enriquecido"

printf '\nCSV generado: %s\n' "$SALIDA"
printf 'CPU_SVC / MEM_SVC: limits configurados por replica (suma de contenedores).\n'
printf 'CPU_NS / MEM_NS: ResourceQuota limits.cpu / limits.memory del namespace.\n'
printf 'SIN_CUOTA: no existe ResourceQuota limits.* para ese recurso.\n'
