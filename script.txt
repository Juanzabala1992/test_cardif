#!/usr/bin/env bash
# Auditoria de limits (NO requests) en OpenShift.
# Consulta Deployments / DeploymentConfigs y las cuotas ResourceQuota del namespace.
# Solo lectura: no cambia recursos del cluster. Requiere bash, oc y awk.
set -euo pipefail

SERVICIOS=(
  'eligibility-api'
  'eligibility-cancellation-policy-legacy-event-capture'
  'eligibility-event-capture'
  'eligibility-issued-policy-legacy-event-capture'
  'notification-center'
  'claims-document-service'
  'customer-portal'
  'customer-portal-backend'
)

SALIDA="${1:-limites_microservicios.csv}"
for cmd in oc awk; do
  command -v "$cmd" >/dev/null 2>&1 || { echo "ERROR: Falta el comando $cmd" >&2; exit 1; }
done
oc whoami >/dev/null 2>&1 || { echo 'ERROR: Primero inicia sesion con oc login.' >&2; exit 1; }

tmpdir="$(mktemp -d)"
trap 'rm -rf "$tmpdir"' EXIT
workloads="$tmpdir/workloads.tsv"
quotas="$tmpdir/quotas.tsv"
filas="$tmpdir/reporte.tsv"
: > "$workloads"; : > "$quotas"; : > "$filas"

# Una linea por contenedor: namespace, deployment, tipo, replicas, contenedor, cpu limit, memoria limit.
# Si un limite no esta configurado: SIN_LIMIT.
plantilla_workload='{{range .items}}{{$w := .}}{{range .spec.template.spec.containers}}{{printf "%s\t%s\t%s\t%v\t%s\t" $w.metadata.namespace $w.metadata.name $w.kind $w.spec.replicas .name}}{{with .resources.limits}}{{with index . "cpu"}}{{.}}{{else}}SIN_LIMIT{{end}}{{else}}SIN_LIMIT{{end}}{{printf "\t"}}{{with .resources.limits}}{{with index . "memory"}}{{.}}{{else}}SIN_LIMIT{{end}}{{else}}SIN_LIMIT{{end}}{{printf "\n"}}{{end}}{{end}}'

if ! oc get deployments -A -o "go-template=$plantilla_workload" > "$workloads"; then
  echo 'ERROR: No se pudieron consultar los Deployments de todos los namespaces.' >&2
  exit 1
fi
# Los DeploymentConfigs son opcionales: algunas instalaciones ya no los usan.
if oc get deploymentconfigs -A -o "go-template=$plantilla_workload" >> "$workloads" 2> "$tmpdir/dc_error"; then
  :
else
  echo 'AVISO: No fue posible consultar DeploymentConfigs; se reportan Deployments.' >&2
fi

# ResourceQuota: hard limits de CPU y memoria en todo el namespace.
# NO equivalen al maximo por contenedor de LimitRange.
plantilla_quota='{{range .items}}{{printf "%s\t%s\t" .metadata.namespace .metadata.name}}{{with .spec.hard}}{{with index . "limits.cpu"}}{{.}}{{else}}SIN_CUOTA{{end}}{{else}}SIN_CUOTA{{end}}{{printf "\t"}}{{with .spec.hard}}{{with index . "limits.memory"}}{{.}}{{else}}SIN_CUOTA{{end}}{{else}}SIN_CUOTA{{end}}{{printf "\n"}}{{end}}'
quota_ok=1
if ! oc get resourcequotas -A -o "go-template=$plantilla_quota" > "$quotas"; then
  echo 'AVISO: No se pudieron consultar las ResourceQuotas; verifique permisos.' >&2
  quota_ok=0
  : > "$quotas"
fi

# Un ciclo FOR sobre cada microservicio de la imagen.
# Al haber mas de un contenedor, se suman sus limits por replica en m y Mi.
for servicio in "${SERVICIOS[@]}"; do
  awk -F '\t' -v OFS='\t' -v objetivo="$servicio" '
    function cpu_m(x) {
      if (x ~ /n$/) return (substr(x, 1, length(x)-1)+0)/1000000
      if (x ~ /u$/) return (substr(x, 1, length(x)-1)+0)/1000
      if (x ~ /m$/) return substr(x, 1, length(x)-1)+0
      return (x+0)*1000
    }
    function mem_mi(x, n) {
      n=x+0
      if (x ~ /Ki$/) return n/1024
      if (x ~ /Mi$/) return n
      if (x ~ /Gi$/) return n*1024
      if (x ~ /Ti$/) return n*1048576
      if (x ~ /Pi$/) return n*1073741824
      if (x ~ /Ei$/) return n*1099511627776
      if (x ~ /[kK]$/) return n*1000/1048576
      if (x ~ /M$/) return n*1000000/1048576
      if (x ~ /G$/) return n*1000000000/1048576
      if (x ~ /T$/) return n*1000000000000/1048576
      if (x ~ /P$/) return n*1000000000000000/1048576
      if (x ~ /E$/) return n*1000000000000000000/1048576
      return n/1048576  # Si no hay sufijo, son bytes
    }
    $2 == objetivo {
      k=$1 SUBSEP $2 SUBSEP $3
      if (!(k in existe)) { existe[k]=1; ns[k]=$1; svc[k]=$2; tipo[k]=$3; replicas[k]=$4 }
      ncont[k]++
      if ($6 == "SIN_LIMIT" || $6 == "" || $6 == "<no value>") sincpu[k]=1
      else { totalcpu[k]+=cpu_m($6); originalcpu[k]=$6 }
      if ($7 == "SIN_LIMIT" || $7 == "" || $7 == "<no value>") sinmem[k]=1
      else { totalmem[k]+=mem_mi($7); originalmem[k]=$7 }
    }
    END {
      for (k in existe) {
        cpu=sincpu[k] ? "SIN_LIMIT" : (ncont[k]==1 ? originalcpu[k] : sprintf("%gm", totalcpu[k]))
        mem=sinmem[k] ? "SIN_LIMIT" : (ncont[k]==1 ? originalmem[k] : sprintf("%gMi", totalmem[k]))
        print ns[k],svc[k],tipo[k],replicas[k],cpu,mem
      }
    }
  ' "$workloads" > "$tmpdir/servicio.tsv"
  if [[ -s "$tmpdir/servicio.tsv" ]]; then
    cat "$tmpdir/servicio.tsv" >> "$filas"
  else
    printf 'NO_ENCONTRADO\t%s\t-\t-\t-\t-\n' "$servicio" >> "$filas"
    echo "NO ENCONTRADO: $servicio (verifique si el Deployment usa otro nombre)" >&2
  fi
done

# CSV delimitado por punto y coma (compatible con Excel configurado en Colombia).
# Cuando hay varias ResourceQuota se listan por nombre, NO se suman sus cuotas.
awk -F '\t' -v OFS=';' -v QUOTA_OK="$quota_ok" '
  function csv(s) { gsub(/"/, "\"\"", s); return "\"" s "\"" }
  FILENAME == ARGV[1] {
    if ($3 != "SIN_CUOTA" && $3 != "") {
      v=$2 ":" $3
      if ($1 in qcpu) qcpu[$1]=qcpu[$1] " | " v; else qcpu[$1]=v
    }
    if ($4 != "SIN_CUOTA" && $4 != "") {
      v=$2 ":" $4
      if ($1 in qmem) qmem[$1]=qmem[$1] " | " v; else qmem[$1]=v
    }
    next
  }
  {
    namespace=$1
    cpuq=(namespace == "NO_ENCONTRADO" ? "-" : (QUOTA_OK == 0 ? "NO_CONSULTADO" : (namespace in qcpu ? qcpu[namespace] : "SIN_CUOTA")))
    memq=(namespace == "NO_ENCONTRADO" ? "-" : (QUOTA_OK == 0 ? "NO_CONSULTADO" : (namespace in qmem ? qmem[namespace] : "SIN_CUOTA")))
    print csv($1),csv($2),csv($3),csv($4),csv($5),csv($6),csv(cpuq),csv(memq)
  }
' "$quotas" "$filas" | {
  printf 'NAMESPACE;MICROSERVICIO;TIPO;REPLICAS;CPU_LIMIT_SERVICIO;MEM_LIMIT_SERVICIO;CPU_LIMIT_NAMESPACE;MEM_LIMIT_NAMESPACE\n'
  sort -t ';' -k2,2 -k1,1
} > "$SALIDA"

# Salida en tabla en consola, con todas las columnas solicitadas.
printf '\n%-28s %-55s %-16s %8s %17s %17s %-25s %-25s\n' \
  'NAMESPACE' 'MICROSERVICIO' 'TIPO' 'REPLICAS' 'CPU LIM SERVICIO' 'MEM LIM SERVICIO' 'CPU LIM NAMESPACE' 'MEM LIM NAMESPACE'
awk -F ';' 'NR>1 {
  for(i=1;i<=NF;i++){gsub(/^"|"$/, "", $i);gsub(/""/, "\"", $i)}
  printf "%-28s %-55s %-16s %8s %17s %17s %-25s %-25s\n",$1,$2,$3,$4,$5,$6,$7,$8
}' "$SALIDA"
printf '\nReporte CSV creado: %s\n' "$SALIDA"
