NS=col-c-uat-digitalsales

oc get route webview-agrario -n $NS \
  -o jsonpath='{.spec.tls.caCertificate}' \
  > ca-actual.pem


oc get route webview-agrario -n $NS \
  -o jsonpath='{.spec.tls.destinationCACertificate}' \
  > destination-ca-actual.pem

openssl x509 -in ca-actual.pem -noout -subject -issuer -dates
openssl x509 -in destination-ca-actual.pem -noout -subject -issuer -dates

openssl x509 \
  -in webview-uat.cardif.com.co.crt \
  -pubkey -noout |
openssl pkey -pubin -outform DER |
sha256sum

openssl pkey \
  -in webview-uat.cardif.com.co.key \
  -pubout -outform DER |
sha256sum

openssl x509 \
  -in webview-uat.cardif.com.co.crt \
  -noout \
  -subject \
  -issuer \
  -dates \
  -ext subjectAltName

openssl verify \
  -CAfile ca-actual.pem \
  webview-uat.cardif.com.co.crt


openssl x509 -in server.crt \
  -checkhost webview-uat.cardif.com.co \
  -noout

# 2. Ver cuántos certificados trae el CRT
grep -c "BEGIN CERTIFICATE" server.crt

# 3. Comprobar la cadena con el CA actual
openssl verify \
  -partial_chain \
  -CAfile ca-actual.pem \
  server.crt
