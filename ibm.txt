CH="$HOME/apigee-rhcos-prod-por-release/datastore"

if [ -s "$CH/apigee-latam-prd-apigee-cassandra.json" ] &&
   [ -f "$CH/Chart.yaml" ]; then

  helm upgrade datastore "$CH" \
    -n apigee \
    --reuse-values \
    --set-string nodeSelector.apigeeData.value=apigee-data-rhcos \
    --dry-run=server \
    --hide-secret \
    > "$HOME/cassandra-dryrun.log" 2>&1 \
    && echo "SIMULACION DATASTORE OK" \
    || echo "SIMULACION FALLO: revisar log protegido"
else
  echo "FALTA ARCHIVO ORIGINAL: NO APLICAR"
fi
