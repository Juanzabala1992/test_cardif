echo "=== RELEASES HELM ==="
helm list -n apigee -o json |
jq -r '
.[] | select(.status=="deployed")
| [.name,.chart] | @tsv'

echo "=== SELECTORES DATASTORE ==="
helm get values datastore -n apigee -o json |
jq '.nodeSelector'

echo "=== ESTRATEGIA CASSANDRA ==="
oc get sts apigee-cassandra-default \
  -n apigee -o json |
jq '{
  strategy: .spec.updateStrategy,
  replicas: .spec.replicas,
  affinity: .spec.template.spec.affinity.nodeAffinity
}'
