echo "=== BUSCAR JSON ORIGINAL ==="

find "$HOME/apigee-rhcos-prod-por-release" \
     "$HOME/apigee-rhcos-prod-charts" \
     -type f \
     -name 'apigee-latam-prd-apigee-cassandra.json' \
     -size +0c \
     -print 2>/dev/null




CH="$HOME/apigee-rhcos-prod-por-release/datastore"

if [ -s "$CH/apigee-latam-prd-apigee-cassandra.json" ] &&
   [ -f "$CH/Chart.yaml" ]; then

  helm upgrade datastore "$CH" \
    -n apigee \
    --reuse-values \
    --set-string nodeSelector.apigeeData.value=apigee-data-rhcos \
    --dry-run=server \
    --hide-secret \
    > "$HOME/cassandra-dryrun.log" 2>&1 \
    && echo "SIMULACION DATASTORE OK" \
    || echo "SIMULACION FALLO: revisar log protegido"
else
  echo "FALTA ARCHIVO ORIGINAL: NO APLICAR"
fi



*********************************


echo "=== HELM DATASTORE ==="
helm status datastore -n apigee

echo "=== SELECTOR ACTUAL ==="
helm get values datastore -n apigee -o json |
jq -r '.nodeSelector.apigeeData.value'

echo "=== RESULTADO SIMULACION ==="
grep -Ei 'error|failed' \
  "$HOME/cassandra-dryrun.log" | head -10
