echo "=== ESTADO CASSANDRA ==="
oc get pods -n apigee -o wide | grep apigee-cassandra-default

echo "=== CONTROLADOR ==="
oc get sts apigee-cassandra-default -n apigee \
  -o json | jq '{
    owners: .metadata.ownerReferences,
    ready: .status.readyReplicas,
    replicas: .spec.replicas
  }'

echo "=== APIGEE DATASTORE ==="
oc get apigeedatastores -n apigee

echo "=== INSTALACIONES HELM ==="
helm list -n apigee

echo "=== VALORES DATASTORE ==="
helm get values datastore -n apigee -o json |
jq '{
  nodeSelector: .nodeSelector,
  cassandraNodeSelector: .cassandra.nodeSelector
}'
