echo "=== NODOS ANTIGUOS ==="
oc get nodes 10.30.12.5 10.30.28.5 10.30.44.5

echo "=== PODS RESTANTES (EXCLUYE DAEMONSETS) ==="
for N in 10.30.12.5 10.30.28.5 10.30.44.5; do
  oc get pods -A \
    --field-selector spec.nodeName="$N" \
    -o json | jq -r --arg n "$N" '
    .items[]
    | select(
        any(.metadata.ownerReferences[]?;
            .kind == "DaemonSet") | not
      )
    | "\($n) \(.metadata.namespace)/\(.metadata.name)"
  '
done
