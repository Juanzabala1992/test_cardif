echo "=== AFINIDAD REAL RUNTIME Y SYNCHRONIZER ==="
oc get pods -n apigee -o json | jq -r '
.items[]
| select(.metadata.name |
    test("^apigee-(runtime|synchronizer)-apigee-latam-np-dev-cl-mtls-"))
| . as $p
| ([.spec.affinity.nodeAffinity
     .requiredDuringSchedulingIgnoredDuringExecution
     .nodeSelectorTerms[]?.matchExpressions[]?
    | select(.key=="apigee-nodepool" or
             .key=="ibm-cloud.kubernetes.io/worker-pool-name")
    | "\(.key)=\(.values|join(","))"
   ] | join(";")) as $aff
| [$p.metadata.name,
   ($p.spec.nodeName // "-"),
   (if $aff=="" then "SIN_AFINIDAD" else $aff end)]
| @tsv'
