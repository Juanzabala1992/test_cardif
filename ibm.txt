      | ($v.env // "") as $env
      | (($v.envs // []
          | map(select(.name == $env))
          | .[0]) // {}) as $e
      | [
          $r,
          $env,
          ($e.serviceAccountPaths.runtime // "SIN_RUTA"),
          ($e.serviceAccountPaths.synchronizer // "SIN_RUTA"),
          ($e.serviceAccountSecretRefs.runtime // "-"),
          ($e.serviceAccountSecretRefs.synchronizer // "-")
        ] | @tsv
    ' >> env-cred-audit.tsv
