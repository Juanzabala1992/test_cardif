oc get pods -n openshift-marketplace \
  -o json | jq -r '
.items[]
| select(.status.phase=="Pending")
| {
    pod: .metadata.name,
    selector: .spec.nodeSelector,
    affinity: .spec.affinity.nodeAffinity
  }'


*******************************

echo "=== CATALOGOS ==="
oc get catalogsources -n openshift-marketplace \
  -o json | jq -r '
.items[]
| [
    .metadata.name,
    (.spec.grpcPodConfig.nodeSelector // {} | tojson)
  ] | @tsv'

echo "=== CONFIGURACION OPERATORHUB ==="
oc get operatorhub cluster -o json |
jq '.spec | {
  disableAllDefaultSources,
  sources
}'

echo "=== WORKERS ==="
oc get nodes \
  -L node-role.kubernetes.io/worker

  -n openshift-marketplace -o yaml \
  > "$HOME/certified-operators-1-backup.yaml"

oc patch catalogsource certified-operators-1 \
  -n openshift-marketplace \
  --type=json \
  -p '[{"op":"replace","path":"/spec/grpcPodConfig/nodeSelector","value":{"kubernetes.io/os":"linux","node-role.kubernetes.io/worker":""}}]'




oc get pods -n openshift-marketplace -o wide

oc get catalogsource certified-operators-1 \
  -n openshift-marketplace \

P=$(oc get pods -n openshift-marketplace -o json |
jq -r '.items[].metadata.name |
select(startswith("certified-operators-1-"))' |
head -1)

echo "=== ERROR DEL CONTENEDOR ==="
oc logs "$P" \
  -n openshift-marketplace \
  --previous --tail=50

echo "=== EVENTOS ==="
oc describe pod "$P" \
  -n openshift-marketplace |
sed -n '/Events:/,$p' | tail -20

echo "=== ESTADO CONTENEDOR ==="
oc get pod "$P" \
  -n openshift-marketplace -o json |
jq '.status.containerStatuses'
