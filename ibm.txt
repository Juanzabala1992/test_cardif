#!/usr/bin/env bash
# PREFLIGHT RHEL 8 -> RHCOS, IBM Cloud OpenShift PROD - salida en PANTALLAS.
# VERSION 3.0, 2026-10-19
# SOLO LECTURA. No crea archivos, no guarda reportes, no crea nodos,
# no hace cordon/drain, no modifica objetos/Helm ni accede al contenido de Secrets.
# Debe ejecutarse con ibmcloud y oc YA configurados para el cluster deseado.
set -uo pipefail
export LC_ALL=C

CL="${CL:-${1:-}}"
APNS="${APNS:-}"               # Namespace de Apigee (opcional; se intenta detectar)
POOL="${POOL:-}"               # Opcional para detallar un pool concreto
NEW_POOL="${NEW_POOL:-}"       # Solo referencia, NO CREA pool
VPC_ID="${VPC_ID:-}"           # Si se conoce, permite consultar detalles de la VPC
export VPC_ID
PAGINA="${PAGINA:-TODAS}"      # TODAS o 1..10
FILAS="${FILAS:-13}"          # Lineas de informacion por fotografia
RUN_NODETOOL="${RUN_NODETOOL:-1}" # nodetool status: solo lectura
LIMITE="${LIMITE:-0}"         # No limita filas: el paginador muestra TODOS los resultados

usage() {
  cat <<'EOF'
Uso: CL='cluster-PROD' APNS=apigee bash Preflight_RHCOS_PROD_PANTALLAS.sh

Muestra textos en la terminal y PAUSA para tomar fotos; no genera ningun archivo.
  PAGINA=4    Muestra solo el bloque 4 (validacion de recursos).
  POOL=nombre PAGINA=2  Detalles del pool indicado.
  VPC_ID=... PAGINA=3  Detalles opcionales de VPC.
  FILAS=10     Muestra solo 10 lineas por foto (pantalla pequena).
  RUN_NODETOOL=0  No ejecuta el comando de diagnostico de Cassandra.

Bloques: 1 contexto, 2 pools, 3 VPC/subredes, 4 nodos/capacidad,

EOF
}
if [[ "$CL" == "-h" || "$CL" == "--help" ]]; then usage; exit 0; fi

# Emite SOLO el texto filtrado del comando. Ninguna salida se almacena en archivos.
raw() {
  local titulo="$1" rc=0 out; shift
  printf '\n-- %s --\n' "$titulo"
  out="$("$@" 2>&1)" || rc=$?
  if ((rc)); then
    printf '[NO DISPONIBLE, rc=%d] %s\n' "$rc" "$(printf '%s\n' "$out" | head -n 2)"
  elif [[ -n "$out" ]]; then printf '%s\n' "$out";
  else echo '(sin resultados)'; fi
}
json() {
  local titulo="$1" filtro="$2" rc=0 out; shift 2
  printf '\n-- %s --\n' "$titulo"
  out="$("$@" 2>&1)" || rc=$?
  if ((rc)); then
    printf '[NO DISPONIBLE, rc=%d] %s\n' "$rc" "$(printf '%s\n' "$out" | head -n 2)"
    return 0
  fi
  if ! jq -e . >/dev/null 2>&1 <<< "$out"; then
    echo '[SIN JSON] La herramienta devolvio datos no compatibles.'
    return 0
  fi
  jq -r "$filtro" <<< "$out" 2>&1 || echo '[ERROR de parseo jq; enviar fotografia]'
}

# Devuelve pools desde etiquetas reales de Kubernetes.
list_pools() {
  oc get nodes -o json 2>/dev/null | jq -r '
    [.items[]?.metadata.labels["ibm-cloud.kubernetes.io/worker-pool-name"] // empty]
    | unique[]' 2>/dev/null || true
}

b1() {
  echo 'CONTEXTO IBM CLOUD + OPENSHIFT'
  echo "Cluster solicitado: $CL"
  raw 'Target IBM (cuenta, region, resource group)' ibmcloud target
  raw 'API de oc en esta sesion' oc whoami --show-server
  raw 'Contexto Kubernetes' oc config current-context
  json 'Version de OpenShift' '
    .items[]? | "Version deseada="+(.spec.desired.version//"-")+" disponible="+([.status.conditions[]?|select(.type=="Available")|.status][0]//"?")' \
    oc get clusterversion -o json
  raw 'Ficha del cluster IBM (comparar con la cuenta PROD)' ibmcloud oc cluster get --cluster "$CL"
  echo 'VERIFICAR MANUALMENTE: CLI, cuenta, API y cluster deben ser exactamente PROD.'
}

b2() {
  echo 'POOLS IBM - OS, CIFRADO KMS/CRK, CAPACIDAD, LABELS'
  raw 'Todos los pools' ibmcloud oc worker-pool ls --cluster "$CL"
  local p pools
  if [[ -n "$POOL" ]]; then
    pools="$POOL"
  else
    pools="$(list_pools)"
  fi
  if [[ -z "$pools" ]]; then
    echo '[AVISO] No se pudieron detectar los nombres de pools.'
    echo 'Usar POOL=nombre_exacto PAGINA=2 para leer su detalle.'
    return 0
  fi
  while IFS= read -r p; do
    [[ -z "$p" ]] && continue
    raw "Pool: $p (incluye OS, flavor, KMS, CRK, labels)" ibmcloud oc worker-pool get --cluster "$CL" --worker-pool "$p"
    raw "Zonas y subredes: $p" ibmcloud oc worker-pool zones --cluster "$CL" --worker-pool "$p"
  done <<< "$pools"
  raw 'Estado de aprovisionamiento workers IBM' ibmcloud oc worker ls --cluster "$CL"
}

b3() {
  echo 'RED / VPC / SUBREDES / SECURITY GROUPS'
  raw 'Cluster IBM - recursos de red' ibmcloud oc cluster get --cluster "$CL" --show-resources
  local p pools
  pools="$(list_pools)"
  while IFS= read -r p; do
    [[ -z "$p" ]] && continue
    raw "Subredes asociadas al pool: $p" ibmcloud oc worker-pool zones --cluster "$CL" --worker-pool "$p"
  done <<< "$pools"
  if [[ -n "$VPC_ID" ]]; then
    json "VPC $VPC_ID - datos clave" '
      {id,name,status,resource_group:(.resource_group.id//"-"),
       address_prefix_management,classic_access,default_security_group:(.default_security_group.id//"-")}' \
      ibmcloud is vpc "$VPC_ID" --output JSON
    json 'Subredes de esta VPC (nombre / id / zona / CIDR / estado)' '
      (if type=="array" then .[] else (.subnets//[])[] end)
      | select((.vpc.id//.vpc_id//"")==env.VPC_ID)
      | [.name,.id,(.zone.name//"-"),(.ipv4_cidr_block//"-"),(.status//"-")] | @tsv' \
      ibmcloud is subnets --output JSON
    json 'Grupos de seguridad de la VPC (nombre/id)' '
      (if type=="array" then .[] else (.security_groups//[])[] end)
      | [.name,.id,(.vpc.id//"-")] | @tsv' \
      ibmcloud is security-groups --vpc "$VPC_ID" --output JSON
    json 'Network ACL vinculadas a VPC (nombre / id)' '
      (if type=="array" then .[] else (.network_acls//[])[] end)
      | select((.vpc.id//.vpc_id//"")==env.VPC_ID)
      | [.name,.id] | @tsv' \
      ibmcloud is network-acls --output JSON
    json 'Tablas de rutas VPC (nombre / id)' '
      (if type=="array" then .[] else (.routing_tables//[])[] end)
      | [.name,.id] | @tsv' \
      ibmcloud is vpc-routing-tables "$VPC_ID" --output JSON
    echo 'Si hay accesos IAM limitados, consultar las reglas SG/ACL por el equipo de red.'
  else
    echo 'Si VPC no aparece: anotar el ID e invocar VPC_ID=<id> PAGINA=3.'
  fi
}

      (.status.readyReplicas//0|tostring)+"/"+(.spec.replicas//1|tostring)]|@tsv' \

b9() {
  echo 'INTEGRATION: APIGEE/CASSANDRA/HELM/BACKUPS'
  raw 'ApigeeDatastores encontrados' oc get apigeedatastores -A
  local ns="$APNS"
  if [[ -z "$ns" ]]; then
    ns="$(oc get apigeedatastores -A -o json 2>/dev/null | jq -r '.items[0].metadata.namespace//empty' 2>/dev/null || true)"
  fi
  if [[ -z "$ns" ]]; then
    echo 'NO SE DETECTO APIGEE: bloque no aplica o define APNS=<namespace>.'
    return 0
  fi
  echo "Namespace Apigee detectado: $ns"
  json 'ApigeeDatastore: status/estado' '
    .items[]|[.metadata.name,
      (.status.state//.status.currentState//.status.phase//"?"),
      (.status.conditions//[]|map(.type+":"+.status)|join(","))]|@tsv' \
    oc get apigeedatastores -n "$ns" -o json
  json 'StatefulSet Cassandra: replicas, strategy y selector' '
    .items[]|select(.metadata.name|test("cassandra";"i"))
    | [.metadata.name, (.status.readyReplicas//0|tostring)+"/"+(.spec.replicas//1|tostring),
      (.spec.updateStrategy.type//"?"),
      (.spec.template.spec.nodeSelector//{}|tojson),
      (.spec.template.spec.affinity.nodeAffinity.requiredDuringSchedulingIgnoredDuringExecution//{}|tojson)]|@tsv' \
    oc get sts -n "$ns" -o json
  json 'Pods Cassandra en nodos actuales' '
    .items[] | select(.metadata.name|test("cassandra";"i"))
    | [.metadata.name,(.status.phase//"?"),(.spec.nodeName//"SIN_NODO"),
       ([.status.containerStatuses[]?|select(.ready)]|length|tostring)+"/"+([.spec.containers[]?]|length|tostring)]|@tsv' \
    oc get pods -n "$ns" -o json
  json 'PDB Apigee' '
    .items[]|[.metadata.name,(.status.disruptionsAllowed//0|tostring)]|@tsv' \
    oc get pdb -n "$ns" -o json
  if command -v helm >/dev/null 2>&1; then
    raw 'Helm releases / version chart' helm list -n "$ns"
    json 'Helm datastore SOLO campos scheduling/backup sin claves' '
      {nodeSelector:(.nodeSelector//{}),
       backup:{enabled:(.cassandra.backup.enabled//.backup.enabled//null),
       schedule:(.cassandra.backup.schedule//null),
       cloudProvider:(.cassandra.backup.cloudProvider//null),
       dbStorageBucket:(.cassandra.backup.dbStorageBucket//null)}}' \
      helm get values datastore -n "$ns" -o json
    raw 'Historial Helm datastore' helm history datastore -n "$ns"
  else
    echo '[FALTA HELM] Consultar release desde la terminal que tenga Helm.'
  fi
  json 'CronJobs Cassandra backup / suspend / plantilla suspend' '
    .items[]|select(.metadata.name|test("cassandra.*backup|backup.*cassandra";"i"))
    | [.metadata.name,(.spec.suspend//false|tostring),
       (.spec.jobTemplate.spec.suspend//false|tostring),
       (.status.lastScheduleTime//"-")]|@tsv' \
    oc get cronjobs -n "$ns" -o json
  json 'Jobs Cassandra backup recientes / completados' '
    [.items[]|select(.metadata.name|test("cassandra.*backup|backup.*cassandra";"i"))]
    | sort_by(.status.completionTime//.metadata.creationTimestamp)
    | .[-12:][]
    | [.metadata.name, (.status.succeeded//0|tostring),
       (.status.failed//0|tostring),(.spec.suspend//false|tostring),
       (.status.completionTime//"-")]|@tsv' \
    oc get jobs -n "$ns" -o json
  if [[ "$RUN_NODETOOL" == "1" ]]; then
    raw 'Cassandra nodetool status (UN en todos)' \
      oc exec -n "$ns" apigee-cassandra-default-0 -c apigee-cassandra -- \
      bash -c 'nodetool -u cassandra -pw "$CASS_PASSWORD" status'
  else
    echo 'Nodetool omitido (RUN_NODETOOL=0).'
  fi
  echo 'ADVERTENCIA: JOB Completed NO demuestra backup externo recuperable.'
}

present() {
  local num="$1" count i from to screen total rawtxt
  local -a lines=()
  rawtxt="$(section "$num")"
  mapfile -t lines <<< "$rawtxt"
  count="${#lines[@]}"
  if ((count==0)); then lines=('Sin resultados'); count=1; fi
  total=$(( (count + FILAS - 1) / FILAS ))
  for (( i=0; i<total; i++ )); do
    from=$((i * FILAS))
    to=$((from + FILAS))
    if ((to>count)); then to=$count; fi
    if [[ -t 0 ]] && command -v clear >/dev/null 2>&1; then clear; fi
    printf '\n====================================================\n'
    printf 'PROD %s | BLOQUE %s/10 | FOTO %s/%s\n' "$CL" "$num" "$((i+1))" "$total"
    printf '====================================================\n'
    printf '%s\n' "${lines[@]:from:to-from}"
    if [[ -t 0 ]]; then
      printf '\n[HAZ FOTO] Enter: siguiente pantalla; Ctrl+C: salir. '
      read -r _ || return 1
    fi
  done
}
