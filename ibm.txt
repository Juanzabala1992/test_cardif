echo "=== CASSANDRA ==="
oc get sts apigee-cassandra-default -n apigee
oc get pods -n apigee -o wide | grep '^apigee-cassandra-default-'

echo "=== PDB ==="
oc get pdb -n apigee | grep -E 'NAME|cassandra'

echo "=== PVC ==="
oc get pvc -n apigee -o wide | grep -i cassandra

echo "=== DISCOS Y ZONAS ==="
oc get pv -o json | jq -r '
.items[]
| select(.spec.claimRef.namespace? == "apigee")
| select((.spec.claimRef.name // "") | contains("cassandra"))
| [
  .metadata.name,
  (.spec.csi.driver // "OTRO"),
  (.spec.storageClassName // "-"),
  .spec.persistentVolumeReclaimPolicy,
  ([.spec.nodeAffinity.required.nodeSelectorTerms[]?
    .matchExpressions[]?
    | select(.key | contains("zone"))
    | .values[]] | join(","))
] | @tsv'

echo "=== SELECTOR DATA ==="
helm get values datastore -n apigee -o json |
jq '.nodeSelector.apigeeData'

ibmcloud oc worker-pool get \
