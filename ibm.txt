#!/usr/bin/env bash
# Integration PROD: procesa un release de Runtime por ciclo y genera un resumen.
# NO migra Cassandra, NO drena workers ni elimina pools.
set -Eeuo pipefail
umask 077

CL='carla-prd-latam-ocw01'
NS='apigee'
OLD='apigee-runtime'
NEW='apigee-runtime-rhcos'
LABEL='ibm-cloud.kubernetes.io/worker-pool-name'
SCRIPT="$HOME/Migracion_RHCOS_Integration_PROD_AUTO_RUNTIME.sh"
DIR="$HOME/apigee-runtime-lote"
STAMP=$(date +%Y%m%d_%H%M%S)
mkdir -p "$DIR"
chmod 700 "$DIR"
LOG="$DIR/ejecucion-$STAMP.log"
REPORT="$DIR/RESUMEN-$STAMP.txt"
CURRENT='ninguno'
STEPS=0
REASON='sin error registrado'

final_report() {
  local result=$?
  trap - EXIT
  set +e
  {
    echo '===== RESULTADO FINAL DE RUNTIME ====='
    echo "Fecha: $(date -Is)"
    echo "Cluster: $CL"
    echo "Releases ejecutados en esta sesion: $STEPS"
    echo "Ultimo release intentado: $CURRENT"
    echo "Motivo de parada: $REASON"
    echo "Codigo de salida: $result"
    echo
    echo '===== ESTADO HELM / SELECTOR ====='
    local helmjson r state sel
    helmjson=$(helm list -n "$NS" -a -o json 2>/dev/null)
    if [[ -z "$helmjson" ]] || ! jq -e 'type=="array"' <<<"$helmjson" >/dev/null 2>&1; then
      echo 'NO SE PUDO CONSULTAR HELM: verificar la conexion.'
    else
      while IFS=$'\t' read -r r state; do
        [[ -n "$r" ]] || continue
        sel=$(helm get values "$r" -n "$NS" -o json 2>/dev/null |
          jq -r '.nodeSelector.apigeeRuntime.value // "SIN_SELECTOR"' 2>/dev/null)
        [[ -n "$sel" ]] || sel='DESCONOCIDO'
        if [[ "$state" != 'deployed' ]]; then
          printf 'ERROR/REVISAR  %-28s Helm=%s selector=%s\n' "$r" "$state" "$sel"
        elif [[ "$sel" == "$NEW" ]]; then
          printf 'SELECTOR_RHCOS %-28s Helm=%s\n' "$r" "$state"
        elif [[ "$sel" == "$OLD" ]]; then
          printf 'PENDIENTE      %-28s Helm=%s\n' "$r" "$state"
        else
          printf 'REVISAR        %-28s Helm=%s selector=%s\n' "$r" "$state" "$sel"
        fi
      done < <(jq -r '.[] | select(.name != "datastore" and (.chart|startswith("apigee-"))) | [.name,.status] | @tsv' <<<"$helmjson")
    fi
    echo
    echo '===== PODS APIGEE QUE AUN ESTAN EN WORKERS RUNTIME RHEL ====='
    local oldnodes oldpods
    oldnodes=$(oc get nodes -l "$LABEL=$OLD" -o json 2>/dev/null | jq -c '[.items[].metadata.name]' 2>/dev/null)
    oldpods=$(oc get pods -n "$NS" -o json 2>/dev/null |
      jq -r --argjson old "${oldnodes:-[]}" '
      .items[]
      | select(.metadata.deletionTimestamp == null)
      | select((.metadata.ownerReferences // [] | any(.kind=="Job")) | not)
      | select((.spec.nodeName // "") as $n | ($old | index($n)) != null)
      | [.metadata.name, .spec.nodeName] | @tsv' 2>/dev/null)
    if [[ -n "$oldpods" ]]; then printf '%s\n' "$oldpods"; else echo 'Ninguno detectado (si la consulta tuvo exito).'; fi
    echo 'IMPORTANTE: SELECTOR_RHCOS no demuestra por si mismo migracion real de los pods.'
  } >"$REPORT"
  cat "$REPORT"
  echo "LOG: $LOG"
  echo "REPORTE: $REPORT"
  return "$result"
}
trap final_report EXIT

command -v flock >/dev/null || { REASON='falta flock'; exit 1; }
exec 9>"$DIR/ejecucion.lock"
flock -n 9 || { REASON='otra ejecucion de lote activa'; exit 1; }
exec > >(tee -a "$LOG") 2>&1

[[ -s "$SCRIPT" ]] || { REASON="no se encontro $SCRIPT"; exit 1; }
bash -n "$SCRIPT" || { REASON='script de Runtime con error de sintaxis'; exit 1; }
grep -q 'AUTO-APLICANDO RUNTIME' "$SCRIPT" || { REASON='script sin AUTO_APPLY_RUNTIME'; exit 1; }
[[ "$(oc config current-context)" == "$CL/"* ]] || { REASON='contexto de OpenShift incorrecto'; exit 1; }

for ((n=1; n<=26; n++)); do
  # No intentar automaticamente un release Helm fallido/pending.
  states=$(helm list -n "$NS" -a -o json) || { REASON='no se puede consultar helm list'; exit 1; }
  invalid=$(jq -r '.[] | select(.name != "datastore" and (.chart|startswith("apigee-")) and .status != "deployed") | .name' <<<"$states")
  [[ -z "$invalid" ]] || { REASON="release Helm no deployed: $invalid"; exit 1; }

  STEPLOG="$DIR/paso-$STAMP-$n.log"
  if [[ -n "${FIN_VENTANA:-}" ]]; then
    limite=$(date -d "$FIN_VENTANA" +%s) || { REASON='FIN_VENTANA no valida'; exit 1; }
    if (( $(date +%s) >= limite )); then
      REASON='termino la ventana autorizada; hay releases pendientes'
      exit 1
    fi
  fi
  echo "===== PASO $n / RUNTIME ====="
  if AUTO_APPLY_RUNTIME=1 bash "$SCRIPT" runtime-next 2>&1 | tee "$STEPLOG"; then
    if grep -q '^HECHO SOLO ' "$STEPLOG"; then
      CURRENT=$(sed -nE 's/^HECHO SOLO ([^ .]+)\..*/\1/p' "$STEPLOG" | tail -1)
      [[ -n "$CURRENT" ]] || { REASON='No se pudo identificar release'; exit 1; }
      sel=$(helm get values "$CURRENT" -n "$NS" -o json |
        jq -r '.nodeSelector.apigeeRuntime.value // empty') || { REASON="fallo helm get values $CURRENT"; exit 1; }
      [[ "$sel" == "$NEW" ]] || { REASON="selector no actualizado para $CURRENT"; exit 1; }
      STEPS=$((STEPS+1))
      echo "COMPLETADO HELM: $CURRENT. Se validara el siguiente release."
    elif grep -q 'No quedan releases con selector antiguo' "$STEPLOG"; then
      CURRENT='ninguno: Runtime sin releases pendientes por selector'
      REASON='todos los selectores Runtime procesados'
      break
    else
      REASON='La salida no confirma ni un release completado ni el fin del proceso'
      exit 1
    fi
  else
    REASON="fallo en el paso $n: revisar $STEPLOG"
    exit 1
  fi
  [[ "$n" -lt 26 ]] || { REASON='limite de pasos alcanzado'; exit 1; }
done
