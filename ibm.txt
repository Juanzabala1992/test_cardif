for N in 10.110.12.4 10.110.28.4 10.110.44.4; do
  echo "========== NODO $N =========="

  oc get node "$N" -o json |
  jq -r '
    "Master: \(.metadata.labels |
      has("node-role.kubernetes.io/master"))",
    "ControlPlane: \(.metadata.labels |
      has("node-role.kubernetes.io/control-plane"))",
    "Pool: \(.metadata.labels[
      "ibm-cloud.kubernetes.io/worker-pool-name"] // "-")"
  '

  echo "=== PODS NO DAEMONSET ==="

  oc get pods -A \
    --field-selector "spec.nodeName=$N" \
    -o json |
  jq -r '
    .items[] |
    select(.status.phase != "Succeeded") |
    select(
      ([.metadata.ownerReferences[]?.kind] |
      index("DaemonSet")) == null
    ) |
    [
      .metadata.namespace,
      .metadata.name,
      (.metadata.ownerReferences[0].kind // "SIN_OWNER"),
      .status.phase
    ] | @tsv
  '
done
