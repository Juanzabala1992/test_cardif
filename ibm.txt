# Revisar selectores de los catálogos
oc get catalogsource -n openshift-marketplace \
  -o custom-columns=NAME:.metadata.name,SELECTOR:.spec.grpcPodConfig.nodeSelector

# Localizar pods del último worker que exigen master
oc get pods -A \
  --field-selector spec.nodeName=10.140.28.19 \
  -o json | jq -r '
.items[] |
select(
  .spec.nodeSelector["node-role.kubernetes.io/master"] != null
  or
  ([.spec.affinity.nodeAffinity.requiredDuringSchedulingIgnoredDuringExecution.nodeSelectorTerms[]?.matchExpressions[]?
    | select(.key=="node-role.kubernetes.io/master")] | length > 0)
) |
[.metadata.namespace,.metadata.name] | @tsv'



oc adm drain 10.140.28.19 \
  --ignore-daemonsets \
  --delete-emptydir-data \
  --timeout=15m
