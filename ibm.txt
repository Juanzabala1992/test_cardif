CL=carla-npr-latam-ocw

for P in apigee-data apigee-data-rhcos; do
  echo "===== POOL: $P ====="

  ibmcloud oc worker-pool get \
    --cluster "$CL" \
    --worker-pool "$P" | grep -Ei \
    'Name:|Flavor:|Operating System:|KMS:|CRK:|KMS Account ID:|Labels:|Workers:'

  ibmcloud oc worker-pool zones \
    --cluster "$CL" \
    --worker-pool "$P"
done

echo "===== CASSANDRA ====="
oc get pods -n apigee -o wide | grep 'apigee-cassandra-default-'
