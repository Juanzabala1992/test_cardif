CRK_ID="<CRK_DEL_POOL_DEFAULT>"
KMS_ACCOUNT="<KMS_ACCOUNT_ID_DEL_POOL_DEFAULT>"



ibmcloud oc worker-pool get \
  --cluster "$CL" \
  --worker-pool default

  --cluster "$CL" \

  --vpc-id "$VPC_ID" \
  --kms-instance "$KMS_INSTANCE" \
  --crk "$CRK_ID" \
  --kms-account-id "$KMS_ACCOUNT"

  --cluster "$CL" \
  --worker-pool "$NEW_POOL"
