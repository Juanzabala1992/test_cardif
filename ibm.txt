echo "=== ZONAS DE LOS PVC ==="
oc get pv -o json | jq -r '
.items[]
| select(
  (.spec.claimRef.namespace // "") == "nexus-poc" or
  (.spec.claimRef.namespace // "") == "lam-c-uat-devops"
)
| [
  .spec.claimRef.namespace,
  .spec.claimRef.name,
  .spec.persistentVolumeReclaimPolicy,
  ([.spec.nodeAffinity.required.nodeSelectorTerms[]?.matchExpressions[]?
    | select(.key | contains("zone"))
    | .values[]] | unique | join(","))
] | @tsv'

echo "=== PODS EN RHEL 8 ==="

| select(

   index("DaemonSet")) == null

  .metadata.namespace,
  .metadata.name,
  .status.phase
] | @tsv'
