# Un Helm upgrade exitoso NO garantiza que los pods ya se hayan movido.

NS='apigee'
OLD_DATA='apigee-data'
OLD_RUNTIME='apigee-runtime'

MODE="${1:-help}"

confirm(){ local expected="$1" text; printf '\nESCRIBE EXACTAMENTE [%s] PARA CONTINUAR: ' "$expected"; read -r text; [[ "$text" == "$expected" ]] || stop 'no se confirmó la operación'; }

check_context(){
  require oc; require ibmcloud; require jq; require helm
  local ctx target server
  ctx=$(oc config current-context) || stop 'no hay contexto OpenShift'
  [[ "$ctx" == "$CL/"* ]] || stop "oc apunta a '$ctx' y no a $CL (ibmcloud oc cluster config --cluster $CL --admin)"
  target=$(ibmcloud target 2>&1) || stop 'no hay sesión IBM Cloud válida'
  grep -Fq "$ACCOUNT" <<<"$target" || stop "la cuenta IBM Cloud no es $ACCOUNT"
  grep -Eq 'Region:[[:space:]]*br-sao([[:space:]]|$)' <<<"$target" || stop 'la región IBM no es br-sao; ejecutar ibmcloud target -r br-sao'
  server=$(oc whoami --show-server) || stop 'oc no conectado'
  [[ "$server" == *'.br-sao.'* ]] || stop "API inesperada: $server"
  say "CUENTA CORRECTA: $ACCOUNT | CLUSTER: $CL | REGION: br-sao"
}

# Evita aceptar pool con VPC/cifrado distintos o con nodos de menos.
check_pool_spec(){
  local old="$1" new="$2" old_info new_info key a b oldz newz
  old_info=$(ibmcloud oc worker-pool get --cluster "$CL" --worker-pool "$old") || stop "no se pudo consultar $old"
  new_info=$(ibmcloud oc worker-pool get --cluster "$CL" --worker-pool "$new") || stop "no existe/no se puede consultar $new"
  grep -Eq '^Operating System:[[:space:]]*RHCOS' <<<"$new_info" || stop "pool $new NO aparece como RHCOS"
  for key in VPC Flavor KMS CRK 'KMS Account ID'; do
    a=$(awk -F: -v k="$key" '$1==k {sub(/^[^:]*:[[:space:]]*/,"");print;exit}' <<<"$old_info")
    b=$(awk -F: -v k="$key" '$1==k {sub(/^[^:]*:[[:space:]]*/,"");print;exit}' <<<"$new_info")
    [[ -n "$a" && -n "$b" ]] || stop "no pude comparar $key entre $old y $new: revisar manualmente antes de proceder"
    [[ "$a" == "$b" ]] || stop "DIFERENCIA $key ($old vs $new): $a / $b"
  done
  oldz=$(ibmcloud oc worker-pool zones --cluster "$CL" --worker-pool "$old" | awk '/^br-sao-[0-9]+[[:space:]]+/{print $1,$2}' | sort)
  newz=$(ibmcloud oc worker-pool zones --cluster "$CL" --worker-pool "$new" | awk '/^br-sao-[0-9]+[[:space:]]+/{print $1,$2}' | sort)
  [[ -n "$oldz" && "$oldz" == "$newz" ]] || stop "las zonas/subredes del pool $new NO coinciden con $old"
  printf '%s -> %s: VPC, flavor, KMS, CRK, cuenta y subredes coinciden\n' "$old" "$new"
}

check_ready_pool(){
  local pool="$1" logical="$2"
  oc get nodes -l "$LABEL_POOL=$pool" -o json | jq -e --arg label "$logical" '
    (.items|length)==3
    and ([.items[]|(.metadata.labels["topology.kubernetes.io/zone"] // .metadata.labels["failure-domain.beta.kubernetes.io/zone"] // "")] | sort) == ["br-sao-1","br-sao-2","br-sao-3"]
    and all(.items[];
      (.metadata.labels["apigee-nodepool"] // "")==$label
      and (.spec.unschedulable != true)
      and ([.status.conditions[] | select(.type=="Ready") | .status] | first)=="True"
      and ((.status.nodeInfo.osImage // "")|contains("CoreOS"))
    )' >/dev/null || stop "pool $pool: hacen falta 3 nodos RHCOS Ready, uno por zona, con apigee-nodepool=$logical"
  printf '%s: 3 workers RHCOS Ready, 3 zonas, etiqueta %s\n' "$pool" "$logical"
}

check_operators(){
  oc get co -o json | jq -e '
    all(.items[];
      ([.status.conditions[]|select(.type=="Available")|.status]|first)=="True" and
      ([.status.conditions[]|select(.type=="Degraded")|.status]|first)!="True"
    )' >/dev/null || stop 'hay operadores de OpenShift no Available o Degraded; revisar oc get co'
}

check_apigee_cr(){
  local type
  for type in apigeedatastore apigeeorg apigeeenv; do
    oc get "$type" -n "$NS" -o json | jq -e '
      (.items|length)>0 and all(.items[];(.status.state // "") == "running")' >/dev/null \
      || stop "algún recurso $type no está running"
  done
}

check_app_pods(){
  local bad
  bad=$(oc get pods -n "$NS" -o json | jq -r '
    .items[]
    | select((.metadata.ownerReferences // [] | any(.kind=="Job")) | not)
    | select(.metadata.deletionTimestamp == null)
    | select(.status.phase != "Running" or
      ((.status.containerStatuses // [])|length == 0) or
      ((.status.containerStatuses // [])|any(.ready != true)))
    | .metadata.name') || stop 'error al consultar pods Apigee'
  [[ -z "$bad" ]] || stop "hay pods Apigee no saludables (no Jobs): $bad"
}

check_data_pvc(){
  oc get pvc -n "$NS" -o json | jq -e '
    [.items[]| select(.metadata.name|startswith("cassandra-data-apigee-cassandra-default-"))] as $p
    | ($p|length) == 3 and all($p[]; .status.phase=="Bound" and (.spec.volumeName//"")!="")
  ' >/dev/null || stop 'los 3 PVC de Cassandra no están Bound'
  oc get pv -o json | jq -e '
    [.items[]|select(.spec.claimRef.namespace=="apigee" and
    (.spec.claimRef.name//""|startswith("cassandra-data-apigee-cassandra-default-")))] as $p
    | ($p|length)==3
    and ([ $p[] | .spec.nodeAffinity.required.nodeSelectorTerms[]?.matchExpressions[]?
      | select(.key|contains("zone")) | .values[] ]|unique|sort)==["br-sao-1","br-sao-2","br-sao-3"]
  ' >/dev/null || stop 'PV Cassandra no mapean uno por zona br-sao-1,2,3. Revisar afinidad manualmente'
}

check_ring(){
  local output want dc1_total dc1_un dc2_total dc2_un
  want=$(oc get sts apigee-cassandra-default -n "$NS" -o jsonpath='{.spec.replicas}') || stop 'StatefulSet Cassandra no encontrado'
  [[ "$want" == 3 ]] || stop "réplicas Cassandra inesperadas: $want (revisar procedimiento)"
  [[ $(oc get sts apigee-cassandra-default -n "$NS" -o jsonpath='{.status.readyReplicas}') == "$want" ]] || stop 'StatefulSet Cassandra no está 3/3 Ready'
  output=$(oc exec -n "$NS" apigee-cassandra-default-0 -c apigee-cassandra -- \
      bash -c 'nodetool -u cassandra -pw "$CASS_PASSWORD" status') || stop 'nodetool status falló'
  say 'ESTADO DEL ANILLO CASSANDRA'; printf '%s\n' "$output"
  # Integration PROD: nodetool devuelve 3 nodos locales dc-1 y 3 remotos dc-2.
  # Contar los 6 como si fueran del StatefulSet local bloquearía el preflight.
  read -r dc1_total dc1_un dc2_total dc2_un < <(awk '
    /^Datacenter:[[:space:]]/ { dc=$2; next }
    $1 ~ /^[UD][NLJM]$/ {
      if (dc=="dc-1") { n1++; if ($1=="UN") u1++ }
      if (dc=="dc-2") { n2++; if ($1=="UN") u2++ }
    }
    END { print n1+0,u1+0,n2+0,u2+0 }
  ' <<<"$output")
  [[ "$dc1_total" == 3 && "$dc1_un" == 3 ]] \
      || stop "dc-1 no está completamente UN (total=$dc1_total UN=$dc1_un; esperado 3/3)"
  [[ "$dc2_total" == 3 && "$dc2_un" == 3 ]] \
      || stop "dc-2 no está completamente UN (total=$dc2_total UN=$dc2_un; esperado 3/3)"
  printf 'Anillo validado: dc-1 %s/%s UN, dc-2 %s/%s UN\n' "$dc1_un" "$dc1_total" "$dc2_un" "$dc2_total"
}

check_preflight(){
  check_context
  say 'VERIFICANDO POOLS NUEVOS / CIFRADO / ZONAS'
  check_pool_spec "$OLD_DATA" "$NEW_DATA"
  check_pool_spec "$OLD_RUNTIME" "$NEW_RUNTIME"
  check_ready_pool "$NEW_DATA" "$OLD_DATA"
  check_ready_pool "$NEW_RUNTIME" "$OLD_RUNTIME"
  check_operators
  check_data_pvc
  check_ring
  check_apigee_cr
  check_app_pods
  say 'PREFLIGHT TECNICO OK. FALTA: aprobación y riesgo de no tener respaldo reciente.'
}

helm_releases(){ helm list -n "$NS" -o json; }
get_release_property(){
  local rel="$1" sel="$2" field="$3"
  helm get values "$rel" -n "$NS" -o json | jq -r --arg s "$sel" --arg f "$field" '.nodeSelector[$s][$f] // ""'
}
get_release_info(){
  local rel="$1"
  helm_releases | jq -er --arg r "$rel" '.[] | select(.name==$r) | [.chart,.status]|@tsv' \
      || stop "no existe el release Helm $rel"
}

order_runtime(){
  helm_releases | jq -r '
    .[] | select(.chart | startswith("apigee-"))
    | select(.chart != "apigee-datastore-1.15.1")
    | [(.chart|sub("-1\\.15\\.1$";"")),.name,.status] | @tsv' |
  awk -F'\t' 'BEGIN{OFS="\t"}
    $1=="apigee-operator" {print "01",$0; next}
    $1=="apigee-redis" {print "02",$0; next}
    $1=="apigee-telemetry" {print "03",$0; next}
    $1=="apigee-ingress-manager" {print "04",$0; next}
    $1=="apigee-org" {print "05",$0; next}
    $1=="apigee-env" {print "06",$0; next}
    $1=="apigee-virtualhost" {print "07",$0; next}
    {print "99",$0}' | sort -k1,1 -k3,3
}

plan(){
  check_context
  say 'PLAN DE RELEASES: CURRENT -> TARGET (sin cambios)'
  printf '%-25s %-28s %-10s %-35s %-28s\n' 'RELEASE' 'CHART' 'HELM' 'CURRENT_SELECTOR' 'PROPUESTA'
  local r chart status k v proposed
  while IFS=$'\t' read -r _ chart r status; do
    k=$(get_release_property "$r" 'apigeeRuntime' key)
    v=$(get_release_property "$r" 'apigeeRuntime' value)
    if [[ "$v" == "$OLD_RUNTIME" && "$k" == "$LABEL_POOL" ]]; then proposed="$NEW_RUNTIME";
    elif [[ "$v" == "$NEW_RUNTIME" && "$k" == "$LABEL_POOL" ]]; then proposed='YA_MIGRADO';
    else proposed='REVISAR_MANUAL'; fi
    printf '%-25s %-28s %-10s %-35s %-28s\n' "$r" "$chart" "$status" "${k:-SIN_KEY}:${v:-SIN_VAL}" "$proposed"
  done < <(order_runtime)
  local dkey dval
  dkey=$(get_release_property datastore 'apigeeData' key)
  dval=$(get_release_property datastore 'apigeeData' value)
  say "DATASTORE actual: $dkey=$dval; destino: $LABEL_POOL=$NEW_DATA"
  say "EL PLAN NO MODIFICA NADA. No migrar DATA si no se acepta formalmente riesgo sin backup."
}

prepare_charts(){
  require helm
  mkdir -m 700 -p "$CHART_HOME"
  local chart
  for chart in apigee-operator apigee-datastore apigee-redis apigee-telemetry apigee-ingress-manager apigee-org apigee-env apigee-virtualhost; do
    if [[ -f "$CHART_HOME/$chart/Chart.yaml" ]]; then
      echo "Existe: $chart"; continue
    fi
    echo "Descargando chart $chart @ $CHART_VERSION"
    helm pull "oci://us-docker.pkg.dev/apigee-release/apigee-hybrid-helm-charts/$chart" \
      --version "$CHART_VERSION" --untar --untardir "$CHART_HOME" || stop "no se pudo descargar $chart"
  done
  say "Charts listos en $CHART_HOME. Los chart que usen .Files.Get requieren sus archivos PROD originales autorizados."
}

healthy_after_upgrade(){
  local rel="$1" count=0
  say "ESPERANDO CONTROLADORES Y APIGEE HEALTHY (máximo 15 min) tras $rel"
  for ((count=0;count<60;count++)); do
    if check_simple_health "$rel"; then echo "Comprobaciones básicas OK para $rel"; return 0; fi
    sleep 15
  done
  stop "estado no saludable tras $rel. DETENER, revisar pods, Helm y pruebas funcionales; NO continuar otros releases"
}
check_simple_health(){
  local rel="$1"
  [[ $(helm status "$rel" -n "$NS" -o json 2>/dev/null | jq -r '.info.status // ""' 2>/dev/null) == 'deployed' ]] || return 1
  oc get co -o json 2>/dev/null | jq -e 'all(.items[]; ([.status.conditions[]|select(.type=="Available")|.status]|first)=="True" and ([.status.conditions[]|select(.type=="Degraded")|.status]|first)!="True")' >/dev/null 2>&1 || return 1
  for type in apigeedatastore apigeeorg apigeeenv; do
    oc get "$type" -n "$NS" -o json 2>/dev/null | jq -e 'all(.items[];(.status.state//"")=="running")' >/dev/null 2>&1 || return 1
  done
  local bad
  bad=$(oc get pods -n "$NS" -o json 2>/dev/null | jq -r '.items[] | select((.metadata.ownerReferences // [] | any(.kind=="Job"))|not) | select(.metadata.deletionTimestamp==null) | select(.status.phase!="Running" or ((.status.containerStatuses//[])|length==0) or ((.status.containerStatuses//[])|any(.ready != true))) | .metadata.name') || return 1
  [[ -z "$bad" ]]
}

apply_release(){
  local rel="$1" which="$2" chartstatus chart status key current target localchart sel="apigeeRuntime"
  [[ "$which" == runtime || "$which" == data ]] || stop "sección inválida: $which"
  if [[ "$which" == data ]]; then
    [[ "$rel" == 'datastore' ]] || stop 'para Cassandra solo release datastore'
    sel='apigeeData'; target="$NEW_DATA"
  else
    [[ "$rel" != 'datastore' ]] || stop 'runtime no debe migrar datastore'
    target="$NEW_RUNTIME"
  fi
  chartstatus=$(get_release_info "$rel")
  chart=${chartstatus%%$'\t'*}
  status=${chartstatus#*$'\t'}
  [[ "$chart" == apigee-*-"$CHART_VERSION" ]] || stop "chart inesperado: $chart"
  [[ "$status" == deployed ]] || stop "release $rel está en estado $status"
  localchart="${chart%-${CHART_VERSION}}"
  [[ -d "$CHART_HOME/$localchart" ]] || stop "falta $CHART_HOME/$localchart (modo 'charts')"
  key=$(get_release_property "$rel" "$sel" key)
  current=$(get_release_property "$rel" "$sel" value)
  [[ "$key" == "$LABEL_POOL" ]] || stop "selector inesperado para $rel: $key"
  if [[ "$current" == "$target" ]]; then echo "$rel YA apunta a $target"; return 0; fi
  local original="$OLD_RUNTIME"
  [[ "$which" == data ]] && original="$OLD_DATA"
  [[ "$current" == "$original" ]] || stop "valor inesperado para $rel: $current"

  # La actualización de datos es mucho más riesgosa, especialmente sin respaldo recuperable.
  if [[ "$which" == data ]]; then
    check_ring
    check_data_pvc
    local active
    active=$(oc get jobs -n "$NS" -o json | jq '[.items[] | select((.metadata.name|startswith("apigee-cassandra-backup-")) and ((.status.active//0)>0))]|length')
    if [[ "$active" -gt 0 ]]; then
      echo "ALERTA: hay $active JOBS ACTIVOS de backup CSI; siguen fallando snapshots."
      confirm 'ACEPTO-JOBS-ACTIVOS'
    fi
    echo 'RIESGO CRITICO: Cassandra PROD sin backup reciente recuperable.'
    echo 'Un rollback Helm o mover los pods de regreso NO recupera datos perdidos.'
    confirm 'ACEPTO-SIN-BACKUP-CASSANDRA'
  fi
  check_operators; check_apigee_cr; check_app_pods; check_data_pvc; check_ring
  mkdir -m 700 -p "$LOG_HOME"
  local drylog="$LOG_HOME/${rel}.dryrun.log"
  say "SIMULACION $rel: $LABEL_POOL=$current -> $LABEL_POOL=$target"
  if ! helm upgrade "$rel" "$CHART_HOME/$localchart" -n "$NS" \
    --reuse-values --set-string "nodeSelector.$sel.value=$target" \
    --dry-run=server --hide-secret > "$drylog" 2>&1; then
    echo "FALLO la simulación. Log protegido: $drylog"
    tail -12 "$drylog" | sed -E 's/(password|secret|private.key).*/[REDACTADO]/Ig' || true
    stop "revisar chart, JSON/Secret PROD requeridos y plantilla antes de aplicar"
  fi
  echo 'Simulación aceptada; revisar en plan de cambio que SOLO deba modificarse nodeSelector.'
  echo 'Un dry-run exitoso por sí solo NO verifica un diff completo ni garantiza disponibilidad.'
  confirm "APLICAR-$rel"
  say "EJECUTANDO un único upgrade: $rel"
  local uplog="$LOG_HOME/${rel}.upgrade.log"
  if ! helm upgrade "$rel" "$CHART_HOME/$localchart" -n "$NS" \
     --reuse-values --set-string "nodeSelector.$sel.value=$target" \
     --timeout 20m > "$uplog" 2>&1; then
    echo "ERROR Helm. Log: $uplog"
    tail -15 "$uplog" | sed -E 's/(password|secret|private.key).*/[REDACTADO]/Ig' || true
    stop 'NO EJECUTAR SIGUIENTES RELEASES NI ROLLBACK AUTOMATICO; revisar estado real primero'
  fi
  healthy_after_upgrade "$rel"
  say "HELM ${rel} HEALTHY; confirmar flujos funcionales antes de avanzar."
  echo "ANTES DE AVANZAR: oc get pods -n $NS -o wide; oc get apigeeenv,apigeeorg,apigeedatastore -n $NS"
}

runtime_next(){
  check_context; check_pool_spec "$OLD_RUNTIME" "$NEW_RUNTIME"; check_ready_pool "$NEW_RUNTIME" "$OLD_RUNTIME"
  check_data_pvc; check_ring
  local r chart status key val
  while IFS=$'\t' read -r _ chart r status; do
    key=$(get_release_property "$r" 'apigeeRuntime' key)
    val=$(get_release_property "$r" 'apigeeRuntime' value)
    if [[ "$key" == "$LABEL_POOL" && "$val" == "$OLD_RUNTIME" ]]; then
      say "SIGUIENTE RELEASE PARA RUNTIME: $r ($chart)"
      apply_release "$r" runtime
      echo "HECHO SOLO $r. VALIDAR APLICACION Y REPETIR runtime-next SI SE APRUEBA."
      return 0
    fi
    if [[ "$key" != "$LABEL_POOL" || ( "$val" != "$NEW_RUNTIME" && "$val" != "$OLD_RUNTIME" ) ]]; then
      echo "ALERTA: $r requiere revisión manual ($key=$val); no puedo asumir que está migrado"
    fi
  done < <(order_runtime)
  echo 'No quedan releases con selector antiguo explícito (verificar los marcados REVISAR_MANUAL mediante plan).'
}

# Antes y despues de mover Cassandra: conserva PVC->PV y Host ID del anillo.
pvc_fingerprint(){
  oc get pvc -n "$NS" -o json | jq -r '
   .items[] | select(.metadata.name|startswith("cassandra-data-apigee-cassandra-default-"))
   | [.metadata.name,.spec.volumeName] | @tsv' | sort
}
ring_host_ids(){
  # Registrar SOLO los 3 Host IDs del dc-1 local; dc-2 pertenece a otro DC.
  oc exec -n "$NS" apigee-cassandra-default-0 -c apigee-cassandra -- \
      bash -c 'nodetool -u cassandra -pw "$CASS_PASSWORD" status' | \
    awk '/^Datacenter:[[:space:]]/ { dc=$2; next } dc=="dc-1" && $1=="UN" {print $(NF-1)}' | sort
}

cassandra(){
  check_context; check_pool_spec "$OLD_DATA" "$NEW_DATA"; check_ready_pool "$NEW_DATA" "$OLD_DATA"
  check_pool_spec "$OLD_RUNTIME" "$NEW_RUNTIME"; check_ready_pool "$NEW_RUNTIME" "$OLD_RUNTIME"
  local oldpvc oldhosts newpvc newhosts
  oldpvc=$(pvc_fingerprint); oldhosts=$(ring_host_ids)
  [[ $(wc -l <<<"$oldpvc") == 3 ]] || stop 'no pude registrar 3 asociaciones PVC/PV antes del movimiento'
  [[ $(wc -l <<<"$oldhosts") == 3 ]] || stop 'no pude registrar 3 Host IDs antes del movimiento'
  apply_release datastore data
  echo 'Esperar reconciliación de ApigeeDatastore. NO BORRAR PODS MANUALMENTE.'
  echo 'Si Cassandra no se mueve: revisar estrategia STS y estado operador. STOP; no repetir Helm a ciegas.'
  say 'ESPERANDO CASSANDRA UN / 3 PODS EN POOL NUEVO (máximo 30 min)'
  local n good=0
  for ((n=0;n<90;n++)); do
    if oc get pods -n "$NS" -o json | jq -e --arg pool "$NEW_DATA" '
      [.items[]|select(.metadata.name|test("^apigee-cassandra-default-[0-9]+$"))] as $p
      | ($p|length)==3
      and all($p[]; .status.phase=="Running" and
         (.status.containerStatuses|all(.ready==true)) and
         .spec.nodeName!="")' >/dev/null 2>&1 &&
       [[ $(oc get pods -n "$NS" -o json | jq -r '[.items[]|select(.metadata.name|test("^apigee-cassandra-default-[0-9]+$"))|.spec.nodeName] | length') == 3 ]] &&
       check_cassandra_placement; then good=1; break; fi
    sleep 20
  done
  [[ "$good" -eq 1 ]] || stop 'Cassandra NO terminó de reprogramarse en el pool RHCOS; NO drenar pool RHEL'
  check_ring
  newpvc=$(pvc_fingerprint); newhosts=$(ring_host_ids)
  [[ "$newpvc" == "$oldpvc" ]] || stop 'HA CAMBIADO la asociación PVC->PV de Cassandra; STOP y revisar'
  [[ "$newhosts" == "$oldhosts" ]] || stop 'CAMBIARON LOS HOST IDS CASSANDRA; STOP, posible pérdida/reinicialización de datos'
  echo 'Cassandra en workers nuevos; dc-1 y dc-2 saludables, mismos PV y Host IDs locales. Validar APIs antes de drenar.' 
}

check_cassandra_placement(){
  local nodes
  nodes=$(oc get nodes -l "$LABEL_POOL=$NEW_DATA" -o json | jq -c '[.items[].metadata.name]') || return 1
  oc get pods -n "$NS" -o json | jq -e --argjson nodes "$nodes" '
    [.items[] | select(.metadata.name|test("^apigee-cassandra-default-[0-9]+$"))] as $p
    | ($p|length)==3
      and all($p[]; (.spec.nodeName // "") as $n | ($nodes | index($n)) != null)
  ' >/dev/null
}

# El drenaje no se permite si cualquier release Apigee conserva la selección antigua.
check_runtime_values_migrated(){
  local r chart status key val pending=0
  while IFS=$'\t' read -r _ chart r status; do
    key=$(get_release_property "$r" apigeeRuntime key)
    val=$(get_release_property "$r" apigeeRuntime value)
    if [[ "$status" != deployed || "$key" != "$LABEL_POOL" || "$val" != "$NEW_RUNTIME" ]]; then
      echo "BLOQUEO RUNTIME: $r ($chart) Helm=$status selector=$key=$val"
      pending=1
    fi
  done < <(order_runtime)
  [[ "$pending" -eq 0 ]] || stop 'existen releases Runtime sin migrar o con afinidad no verificada'
}

drain_node(){
  local node="${1:-}" pool oldn r
  [[ -n "$node" ]] || stop 'debe indicar el nodo RHEL antiguo (no una IP sin comprobar)'
  check_context
  check_pool_spec "$OLD_DATA" "$NEW_DATA"; check_ready_pool "$NEW_DATA" "$OLD_DATA"
  check_pool_spec "$OLD_RUNTIME" "$NEW_RUNTIME"; check_ready_pool "$NEW_RUNTIME" "$OLD_RUNTIME"
  pool=$(oc get node "$node" -o json | jq -r '.metadata.labels["ibm-cloud.kubernetes.io/worker-pool-name"] // ""') || stop 'nodo no encontrado'
  [[ "$pool" == "$OLD_DATA" || "$pool" == "$OLD_RUNTIME" ]] || stop "el nodo $node no pertenece a un pool RHEL antiguo: $pool"
  # Nunca usar drain como método para mover Cassandra con backups no recuperables.
  if oc get pods -n "$NS" -o json | jq -e --arg node "$node" '
    any(.items[]; (.spec.nodeName==$node) and (.metadata.name|test("^apigee-cassandra-default-[0-9]+$")))' >/dev/null; then
    stop 'CASSANDRA AUN EN NODO ANTIGUO. Drenaje bloqueado; mover por operador/Helm con verificación UN.'
  fi
  if [[ "$pool" == "$OLD_RUNTIME" ]]; then
    check_runtime_values_migrated
  else
    [[ $(get_release_property datastore apigeeData value) == "$NEW_DATA" ]] \
      || stop 'datastore aun apunta al pool viejo'
    check_cassandra_placement || stop 'Cassandra aun tiene pods fuera de NEW_DATA'
    check_ring
  fi
  check_operators; check_app_pods
  local active_backups
  active_backups=$(oc get jobs -n "$NS" -o json | jq '[.items[] | select((.metadata.name|startswith("apigee-cassandra-backup-")) and ((.status.active//0)>0))]|length')
  if [[ "$active_backups" -gt 0 ]]; then
    echo "ALERTA: $active_backups jobs Cassandra backup CSI siguen activos (no es un backup verificable)."
    confirm 'ACEPTO-DRENAR-CON-BACKUPS-BLOQUEADOS'
  fi
  echo "DRENAR: $node (pool antiguo $pool). Puede interrumpir servicios y borrar emptyDir temporal."
  echo 'Se respetarán PDB; NO se usarán --force ni --disable-eviction.'
  confirm "DRENAR-$node"
  oc adm drain "$node" --ignore-daemonsets --delete-emptydir-data --timeout=15m \
    || stop 'drain bloqueado/timeout; NO eliminar worker ni ignorar PDB. Revisar salida.'
  say 'DRENADO; verificando operadores'
  check_operators
  oc get nodes
  echo 'Comprobar redis/runtime/synchronizer e ingreso de APIs antes del siguiente nodo.'
}

status(){
  check_context
  say 'IBM POOLS'; ibmcloud oc worker-pool ls --cluster "$CL"
  say 'NODOS Y POOLS'; oc get nodes -L "$LABEL_POOL" -L topology.kubernetes.io/zone
  say 'APIGEE DATASTORE / CASSANDRA'; oc get apigeedatastores -n "$NS"
  oc get sts apigee-cassandra-default -n "$NS"
  oc get pods -n "$NS" -o wide | grep -E 'apigee-cassandra-default-[0-9]+|apigee-redis-default' || true
  say 'APIGEE ORG Y ENVS'; oc get apigeeorg,apigeeenv -n "$NS" || true
  say 'PODS APIGEE RESIDUALES EN RHEL (EXCLUYE JOBS)'
  oc get pods -n "$NS" -o json | jq -r '
   .items[] | select((.metadata.ownerReferences//[]|any(.kind=="Job"))|not)
   | [.spec.nodeName,.metadata.name,.status.phase] | @tsv' | while IFS=$'\t' read -r node pod phase; do
    [[ -n "$node" ]] || continue
    local p
    p=$(oc get node "$node" -o json | jq -r '.metadata.labels["ibm-cloud.kubernetes.io/worker-pool-name"]//""') || continue
    if [[ "$p" == "$OLD_DATA" || "$p" == "$OLD_RUNTIME" ]]; then
      printf '%s\t%s\t%s\t%s\n' "$node" "$p" "$pod" "$phase"
    fi
  done
  say 'PDB sin interrupciones permitidas'
  oc get pdb -A -o json | jq -r '.items[]|select((.status.disruptionsAllowed//0)==0)|[.metadata.namespace,.metadata.name]|@tsv' | head -30
  say 'OPERADORES'; oc get co | awk 'NR==1 || $3!="True" || $5!="False"'
  echo 'NOTA: revisar pruebas de negocio y respaldo ausente. NO se borran pools.'
}
