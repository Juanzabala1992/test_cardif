#!/usr/bin/env bash
# Reanuda únicamente los releases apigee-env que conservan la afinidad antigua.
# No actualiza dev-env si Helm está failed, no toca apigee-data ni retira workers.
set -Eeuo pipefail
umask 077
NS=apigee
WORK="$HOME/apigee-migration-1151"
cd "$WORK" || { echo "DETENIDO: falta $WORK"; exit 1; }
[[ -d ./apigee-env ]] || { echo 'DETENIDO: falta el chart apigee-env'; exit 1; }
CRED="$WORK/apigee-env/apigee-latam-npr-apigee-non-prod.json"
[[ ! -e "$CRED" ]] || { echo "DETENIDO: existe JSON temporal anterior en $CRED. Revisarlo manualmente antes de continuar."; exit 1; }
LOG="$WORK/resume-rhcos-$(date +%Y%m%d-%H%M%S)"
mkdir -m 700 "$LOG"
trap 'rm -f "$CRED"' EXIT
trap 'exit 130' INT
trap 'exit 143' TERM

# Precondiciones: no correr mientras otro Helm upgrade está pendiente.
helm list -n "$NS" -o json > "$LOG/helm-releases.json"
TOTAL=$(jq '[.[] | select(.chart | startswith("apigee-env-"))] | length' "$LOG/helm-releases.json")
[[ "$TOTAL" -eq 27 ]] || { echo "DETENIDO: esperaba 27 releases apigee-env; encontré $TOTAL"; exit 1; }
BAD=$(jq -r '.[] | select(.chart | startswith("apigee-env-")) | select(.status != "deployed" and ((.name == "dev-env" and .status == "failed") | not)) | "\(.name)=\(.status)"' "$LOG/helm-releases.json")
[[ -z "$BAD" ]] || { printf 'DETENIDO: Helm en estado inesperado:\n%s\n' "$BAD"; exit 1; }
[[ $(oc get nodes --no-headers | awk '$2!="Ready"{n++}END{print n+0}') -eq 0 ]] || { echo 'DETENIDO: hay nodos No Ready'; exit 1; }
oc get apigeeorg -n "$NS" -o json | jq -e 'all(.items[]; .status.state=="running")' >/dev/null || { echo 'DETENIDO: ApigeeOrg no está running'; exit 1; }
oc get apigeeenv -n "$NS" -o json | jq -e 'all(.items[]; .status.state=="running")' >/dev/null || { echo 'DETENIDO: hay ApigeeEnv no running'; exit 1; }
[[ $(oc get pods -n "$NS" -o json | jq '[.items[] | select(.status.phase=="Pending")] | length') -eq 0 ]] || { echo 'DETENIDO: pods Pending en Apigee'; exit 1; }

oc get secrets -n "$NS" -o json | jq -r '.items[].metadata.name' > "$LOG/secret-names.txt"
oc get apigeeenv -n "$NS" -o json | jq -r '.items[].metadata.name' > "$LOG/env-names.txt"
: > "$LOG/plan.tsv"

# Incluye solamente releases desplegados con la clave de afinidad anterior.
while IFS=$'\t' read -r R ST; do
  helm get values "$R" -n "$NS" -o json > "$LOG/$R.values.json"
  KEY=$(jq -r '.nodeSelector.apigeeRuntime.key // "-"' "$LOG/$R.values.json")
  VAL=$(jq -r '.nodeSelector.apigeeRuntime.value // "-"' "$LOG/$R.values.json")
  [[ "$VAL" == 'apigee-runtime' ]] || { echo "DETENIDO: selector inesperado en $R"; exit 1; }
  case "$KEY" in
    apigee-nodepool)
      echo "YA ACTUALIZADO $R" ;;
    'ibm-cloud.kubernetes.io/worker-pool-name')
      if [[ "$ST" == 'failed' && "$R" == 'dev-env' ]]; then
        echo 'EXCLUIDO dev-env (Helm failed)'
        continue
      fi
      [[ "$ST" == 'deployed' ]] || { echo "DETENIDO: $R tiene estado $ST"; exit 1; }
      ENV=$(jq -r '.env // empty' "$LOG/$R.values.json")
      [[ "$ENV" =~ ^[a-z0-9-]+$ ]] || { echo "DETENIDO: env inválido en $R"; exit 1; }
      mapfile -t S < <(grep -E "^apigee-(runtime|synchronizer|udca)-apigee-latam-np-${ENV}-[0-9a-f]{7}-svc-account$" "$LOG/secret-names.txt" || true)
      [[ ${#S[@]} -eq 3 ]] || { echo "DETENIDO: $R, ${#S[@]} Secrets, esperaba 3"; exit 1; }
      H0=''
      for X in "${S[@]}"; do
        H=$(oc get secret "$X" -n "$NS" -o json | jq -er '.data["client_secret.json"]' | base64 -d | sha256sum | cut -d' ' -f1)
        [[ -z "$H0" || "$H0" == "$H" ]] || { echo "DETENIDO: credenciales diferentes en $R"; exit 1; }
        H0="$H"
      done
      CR=$(grep -E "^apigee-latam-np-${ENV}-[a-f0-9]{7}$" "$LOG/env-names.txt" || true)
      [[ -n "$CR" && "$CR" != *$'\n'* ]] || { echo "DETENIDO: ApigeeEnv no único en $R"; exit 1; }
      printf '%s\t%s\t%s\t%s\n' "$R" "$ENV" "$CR" "${S[0]}" >> "$LOG/plan.tsv" ;;
    *) echo "DETENIDO: selector desconocido en $R: $KEY"; exit 1 ;;
  esac
done < <(jq -r '.[] | select(.chart | startswith("apigee-env-")) | [.name,.status] | @tsv' "$LOG/helm-releases.json" | sort)

COUNT=$(wc -l < "$LOG/plan.tsv")
echo "RELEASES PENDIENTES DE ACTUALIZAR: $COUNT"
[[ "$COUNT" -gt 0 ]] || { echo 'No hay releases elegibles pendientes'; exit 0; }
[[ "$COUNT" -le 15 ]] || { echo 'DETENIDO: más de 15 pendientes; revisar antes de continuar'; exit 1; }

create_cred() {
  local secret="$1"
  [[ ! -e "$CRED" ]] || { echo 'DETENIDO: JSON temporal existente'; return 1; }
  oc get secret "$secret" -n "$NS" -o json | jq -er '.data["client_secret.json"]' | base64 -d > "$CRED"
  jq -e 'type=="object"' "$CRED" >/dev/null
}
check_ready() {
  local cr="$1"
  [[ $(oc get apigeeenv "$cr" -n "$NS" -o jsonpath='{.status.state}' 2>/dev/null) == running ]] || return 1
  oc get apigeedeployments -n "$NS" -o json | jq -e --arg cr "$cr" '
    [.items[] | select(.metadata.name == ("apigee-runtime-" + $cr) or .metadata.name == ("apigee-synchronizer-" + $cr))] as $d
    | ($d|length) == 2
      and all($d[]; .status.state=="running" and ((.status.replicas.ready//0) >= 1))' >/dev/null || return 1
  oc get pods -n "$NS" -o json | jq -e --arg cr "$cr" '
    .items as $pods
    | all(["runtime","synchronizer"][]; . as $kind |
      any($pods[];
        (.metadata.name | startswith("apigee-" + $kind + "-" + $cr + "-"))
        and .status.phase == "Running"
        and ((.status.containerStatuses // []) | length > 0)
        and all(.status.containerStatuses[]; .ready == true)
        and any(.spec.affinity.nodeAffinity.requiredDuringSchedulingIgnoredDuringExecution.nodeSelectorTerms[]?.matchExpressions[]?;
          .key == "apigee-nodepool" and (.values | index("apigee-runtime") != null))
      ))' >/dev/null
}

# Etapa 1: simular TODO el plan. No hay cambios reales aquí.
while IFS=$'\t' read -r R ENV CR S; do
  create_cred "$S"
  if ! helm upgrade "$R" ./apigee-env -n "$NS" --reuse-values \
      --set-string "env=$ENV" \
      --set-string nodeSelector.apigeeRuntime.key=apigee-nodepool \
      --set-string nodeSelector.apigeeRuntime.value=apigee-runtime \
      --dry-run=server --hide-secret > "$LOG/$R.dryrun.log" 2>&1; then
    echo "DETENIDO: dry-run falló en $R. Revisar $LOG/$R.dryrun.log"; exit 1
  fi
  rm -f "$CRED"
  echo "PREVALIDADO $R"
done < "$LOG/plan.tsv"

echo "SIMULACIONES OK para $COUNT releases. Ningún cambio real todavía."
read -r -p 'Para aplicar los cambios reales secuencialmente escribe APLICAR: ' CONFIRM
[[ "$CONFIRM" == APLICAR ]] || { echo 'Cancelado antes de aplicar'; exit 0; }

# Etapa 2: revisar estado justo antes de cada release y aplicar secuencialmente.
while IFS=$'\t' read -r R ENV CR S; do
  CURRENT=$(helm get values "$R" -n "$NS" -o json | jq -r '.nodeSelector.apigeeRuntime.key // "-"')
  if [[ "$CURRENT" == apigee-nodepool ]]; then
    echo "OMITIDO $R: actualizado después de generar el plan"
    continue
  fi
  [[ "$CURRENT" == 'ibm-cloud.kubernetes.io/worker-pool-name' ]] || { echo "DETENIDO: selector alterado en $R"; exit 1; }
  STATUS=$(helm status "$R" -n "$NS" -o json | jq -r '.info.status')
  [[ "$STATUS" == deployed ]] || { echo "DETENIDO: $R está en $STATUS"; exit 1; }
  echo "ACTUALIZANDO $R"
  create_cred "$S"
  if ! helm upgrade "$R" ./apigee-env -n "$NS" --reuse-values \
      --set-string "env=$ENV" \
      --set-string nodeSelector.apigeeRuntime.key=apigee-nodepool \
      --set-string nodeSelector.apigeeRuntime.value=apigee-runtime \
      --rollback-on-failure --timeout 15m > "$LOG/$R.upgrade.log" 2>&1; then
    echo "DETENIDO: error al actualizar $R. Revisar $LOG/$R.upgrade.log"; exit 1
  fi
  rm -f "$CRED"
  HEALTHY=false
  for _ in $(seq 1 90); do
    if check_ready "$CR"; then HEALTHY=true; break; fi
    sleep 10
  done
  [[ "$HEALTHY" == true ]] || { echo "DETENIDO: $R no recuperó salud en 15m"; exit 1; }
  [[ $(oc get pods -n "$NS" -o json | jq '[.items[] | select(.status.phase=="Pending")] | length') -eq 0 ]] || { echo "DETENIDO: pods Pending después de $R"; exit 1; }
  echo "OK $R"
done < "$LOG/plan.tsv"

echo 'LOTE REANUDADO COMPLETO. FALTA: dev-env (failed), ingress-manager y retiro controlado de los dos workers RHEL 8.'



***************************

oc get apigeeredis default -n apigee

oc get sts apigee-redis-default -n apigee -o json |

echo "=== WORKERS RHCOS ==="
oc get nodes -l apigee-nodepool=apigee-runtime

echo "=== CAPACIDAD ==="
oc describe nodes -l apigee-nodepool=apigee-runtime |

)

  > dryrun-redis-rhcos.txt 2>&1 || {
    echo "DRY-RUN FALLÓ. No se aplicaron cambios."
    exit 1
  }

echo "DRY-RUN OK"
