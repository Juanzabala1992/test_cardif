echo "=== APIGEE DEPLOYMENTS ==="

for A in \
  apigee-metrics-adapter-apigee-telemetry \
  apigee-metrics-apigee-telemetry-app
do
  echo "=== $A ==="
  oc get ad "$A" -n apigee -o json | jq '{
    state: .status.state,
    replicas: .status.replicas,
    affinity: [
      .spec | .. | objects |
      select(
        .key? == "apigee-nodepool" or
        .key? == "ibm-cloud.kubernetes.io/worker-pool-name"
      ) | {key, values}
    ]
  }'
done

echo "=== REPLICASETS METRICS ==="

oc get rs -n apigee -o json | jq -r '
.items[]
| select(.metadata.name |
    test("^apigee-metrics-(adapter|apigee-telemetry-app)"))
| [
    .metadata.name,
    (.spec.replicas // 0),
    (.status.readyReplicas // 0),
    ([
      .spec.template.spec.affinity.nodeAffinity
      .requiredDuringSchedulingIgnoredDuringExecution
      .nodeSelectorTerms[]?.matchExpressions[]?
      | "\(.key)=\((.values // []) | join(","))"
    ] | join(";"))
  ] | @tsv'
