apiVersion: batch/v1
kind: CronJob
metadata:
  name: restart-car-services
  namespace: chl-c-prd-carboosting
spec:

  # Cada 2 horas
  schedule: "0 */2 * * *"

  # Evita que arranque otro Job si el anterior sigue ejecutándose
  concurrencyPolicy: Forbid

  # Si por algún motivo no pudo arrancar a la hora programada,
  # tiene hasta 10 minutos para hacerlo
  startingDeadlineSeconds: 600

  successfulJobsHistoryLimit: 3
  failedJobsHistoryLimit: 3

  jobTemplate:
    spec:
      backoffLimit: 1
      activeDeadlineSeconds: 900

      template:
        spec:
          serviceAccountName: cron-restart-carboosting
          restartPolicy: Never

          containers:
            - name: restart-services

              # CLI interno del cluster
              image: image-registry.openshift-image-registry.svc:5000/openshift/cli:latest
              imagePullPolicy: IfNotPresent

              # Valores bajos debido al ResourceQuota actual
              resources:
                requests:
                  cpu: 25m
                  memory: 32Mi
                limits:
                  cpu: 50m
                  memory: 64Mi

              command:
                - /bin/bash
                - -c
                - |
                  set -euo pipefail

                  NS="chl-c-prd-carboosting"

                  restart_pod() {

                    PREFIX="$1"

                    echo "=================================================="
                    echo "[$(date)] Reiniciando servicio: ${PREFIX}"
                    echo "=================================================="

                    POD=$(oc get pods -n "${NS}" -o name \
                      | sed 's#pod/##' \
                      | grep "^${PREFIX}-" \
                      | head -n 1 || true)

                    if [ -z "${POD}" ]; then
                      echo "ERROR: No se encontro un pod para ${PREFIX}"
                      return 1
                    fi

                    echo "Pod actual: ${POD}"
                    echo "Eliminando pod..."

                    oc delete pod "${POD}" \
                      -n "${NS}" \
                      --wait=true

                    echo "Pod ${POD} eliminado."
                    echo "Esperando que el nuevo pod quede Running/Ready..."

                    i=0

                    while [ ${i} -lt 90 ]; do

                      NEWPOD=$(oc get pods -n "${NS}" \
                        -o jsonpath='{range .items[*]}{.metadata.name}{" "}{.status.phase}{" "}{range .status.conditions[?(@.type=="Ready")]}{.status}{end}{"\n"}{end}' \
                        | awk -v p="${PREFIX}" \
                          '$1 ~ ("^" p "-") && $2 == "Running" && $3 == "True" {print $1; exit}')

                      if [ -n "${NEWPOD}" ] && [ "${NEWPOD}" != "${POD}" ]; then

                        echo "OK: ${PREFIX} esta nuevamente disponible."
                        echo "Nuevo pod: ${NEWPOD}"
                        echo ""

                        return 0
                      fi

                      i=$((i+1))

                      echo "Esperando ${PREFIX}... intento ${i}/90"

                      sleep 5

                    done

                    echo "ERROR: ${PREFIX} no quedo Ready dentro del tiempo esperado."
                    return 1
                  }


                  echo ""
                  echo "Inicio proceso automatico: $(date)"
                  echo ""

                  restart_pod "cb-car-kafka-quotation"

                  restart_pod "cb-car-homologator"

                  restart_pod "cb-car-quotation"

                  echo ""
                  echo "=================================================="
                  echo "Todos los servicios fueron reiniciados correctamente"
                  echo "Fin: $(date)"
                  echo "=================================================="
