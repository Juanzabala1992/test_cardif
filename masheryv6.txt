NS=lam-c-dev-mashery-v6

S=$(oc get deploy configui-deploy -n "$NS" \
  -o jsonpath='{.spec.template.spec.volumes[?(@.name=="configui-key-vol")].secret.secretName}')

echo "Secret utilizado: $S"

oc get secret "$S" -n "$NS" -o json | python3 -c '
import sys, json, base64
data = json.load(sys.stdin).get("data", {})
for name, value in data.items():
    raw = base64.b64decode(value)
    lines = raw.splitlines()
    first = lines[0] if lines else b""
    if first.startswith(b"-----BEGIN "):
        result = first.decode("ascii", "replace")
    else:
        result = "Formato no PEM o binario"
    if b"ENCRYPTED" in b"\n".join(lines[:4]):
        result += " [CIFRADA]"
    print(name + " => " + result)
'
