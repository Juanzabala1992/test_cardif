NS=lam-c-dev-mashery-v6

for JOB in loader-job-full-29867040 loader-job-onprem-29866860; do
  POD=$(oc get pods -n "$NS" -o name \
    | grep "pod/$JOB-" | head -1 | cut -d/ -f2)

  echo "===== $JOB ====="

  if [ -n "$POD" ]; then
    oc describe pod "$POD" -n "$NS" \
      | sed -n '/Events:/,$p' | tail -n 15
  else
    echo "No se encontro pod"
  fi
done
