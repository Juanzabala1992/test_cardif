NS=lam-c-dev-mashery-v6
umask 077

T=$(mktemp -d /dev/shm/configui-tls.XXXXXX) || exit 1

openssl genpkey -algorithm RSA \
  -pkeyopt rsa_keygen_bits:3072 \
  -out "$T/tml-cm-key.pk8"

openssl req -new -x509 -sha256 -days 30 \
  -key "$T/tml-cm-key.pk8" \
  -out "$T/tml-cm-crt.pem" \
  -subj "/CN=configui-svc.${NS}.svc" \
  -addext "subjectAltName=DNS:configui-svc,DNS:configui-svc.${NS},DNS:configui-svc.${NS}.svc,DNS:configui-svc.${NS}.svc.cluster.local"


****************************************


KEY_HASH=$(openssl pkey \
  -in "$T/tml-cm-key.pk8" \
  -pubout -outform DER | openssl dgst -sha256)

CERT_HASH=$(openssl x509 \
  -in "$T/tml-cm-crt.pem" \
  -pubkey -noout | \
  openssl pkey -pubin -outform DER | \
  openssl dgst -sha256)

if [ "$KEY_HASH" = "$CERT_HASH" ] && [ -n "$KEY_HASH" ]; then
  echo "CERTIFICADO Y CLAVE COINCIDEN"
else
  echo "ERROR: CERTIFICADO Y CLAVE NO COINCIDEN"
fi

head -n 1 "$T/tml-cm-key.pk8"

openssl x509 -in "$T/tml-cm-crt.pem" \
  -noout -subject -dates -ext subjectAltName
